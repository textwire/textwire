import (
        "io/fs"
        "os"
        "path/filepath"
        "time"
)

// DefaultMaxComponentDepth is the default value of Config.MaxComponentDepth.
const DefaultMaxComponentDepth = 500

// Config holds the configuration settings for Textwire template engine.
type Config struct {
        // TemplateDir specifies the directory containing Textwire template files.
        // Note: If TemplatesFS is provided, TemplateDir is ignored because there
        // are no absolute paths for embedded files.
        // Default: "templates"
        TemplateDir string

        // TemplateFS provides an optional fs.FS filesystem for template access.
        // Use this field to embed templates into your binary using Go's embed
        // package. When provided, TemplateDir is not used for file access.
        // Default: os.DirFS(TemplateDir)
        TemplateFS fs.FS

        // TemplateExt defines the file extension for Textwire template files.
        // Note: Using a different extension may disable syntax highlighting
        // in editors like VSCode when using the Textwire extension.
        // Default: ".tw"
        TemplateExt string

        // ErrorPagePath sets the relative path to a custom error page template.
        // The path is relative to the template directory
        // (TemplateDir or TemplatesFS root).
        // Default: embed/default-error-page.tw (default error page)
        ErrorPagePath string

        // DebugMode enables detailed error reporting in the browser and server
        // logs. When true, error messages with file paths and line numbers are
        // displayed during development.
        // Default: false
        DebugMode bool

        // GlobalData stores shared data accessible across all templates.
        // Access these values in templates using the `global` object
        // (e.g., `global.authUser`). Useful for storing environment variables,
        // configuration, or common data.
        GlobalData map[string]any

        // FileWatcher watches all of your template files for changes and
        // automatically re-parses them when they are modified. This is intended
        // for development use only and should not be enabled in production due to
        // performance implications. It doesn't work with TemplateFS configuration
        // enabled! Disable it to use file refresh functionality.
        // Default: false
        FileWatcher bool

        // WatcherInterval specifies how often Textwire checks for changes in
        // template files when FileWatcher is enabled. The higher the interval,
        // the less frequently Textwire checks for file changes, which can reduce
        // CPU usage but may delay updates. Values less than 1 second will be
        // treated as the default (1 second). Adjust this value based on your
        // development needs.
        // Default: time.Second (1 second)
        WatcherInterval time.Duration

        // MaxComponentDepth limits how many @component directives can be
        // rendering inside each other at the same time during one render. It
        // stops a component that includes itself, directly or through other
        // components, with an error instead of crashing the program with a stack
        // overflow. Only nesting counts: components rendered one after another,
        // like inside of a loop, don't add up. Values less than 1 are treated as
        // the default.
        // Default: 500
        MaxComponentDepth int

        // usesFS is a flag to determine if user uses TemplateFS or not.
        usesFS bool
}

func New() *Config {
        return &Config{
                TemplateDir:       "templates",
                TemplateExt:       ".tw",
                ErrorPagePath:     "",
                DebugMode:         false,
                GlobalData:        map[string]any{},
                WatcherInterval:   time.Second,
                MaxComponentDepth: DefaultMaxComponentDepth,
        }
}

// UsesFS returns value of usesFS field since that field is private.
func (c *Config) UsesFS() bool {
        return c.usesFS
}

func (c *Config) Configure(opt *Config) {
        if opt == nil {
                c.TemplateFS = os.DirFS(c.TemplateDir)
                return
        }

        if opt.TemplateDir != "" {
                c.TemplateDir = filepath.Clean(opt.TemplateDir)
        }

        if opt.TemplateExt != "" {
                c.TemplateExt = opt.TemplateExt
        }

        if opt.TemplateFS == nil {
                c.TemplateFS = os.DirFS(c.TemplateDir)
        } else {
                c.TemplateFS = opt.TemplateFS
        }

        if opt.ErrorPagePath != "" {
                c.ErrorPagePath = opt.ErrorPagePath
        }

        if opt.WatcherInterval >= time.Second {
                c.WatcherInterval = opt.WatcherInterval
        }

        if opt.GlobalData != nil {
                c.GlobalData = opt.GlobalData
        }

        if opt.MaxComponentDepth > 0 {
                c.MaxComponentDepth = opt.MaxComponentDepth
        }

        c.FileWatcher = opt.FileWatcher
        c.DebugMode = opt.DebugMode
        c.usesFS = opt.TemplateFS != nil
}

        out io.Writer
}

        return &WatcherLogger{}

        l.print(colorCyan, text)

        l.print(colorRed, text)

        out := l.out
        if out == nil {
                out = os.Stdout
        }

        _, err := fmt.Fprintf(
                out,
                "%s[Watcher]%s %s%s%s\n",
                colorYellow,
                colorReset,
                color,
                text,
                colorReset,
        )

        if err != nil {
                fmt.Fprintf(os.Stderr, "Failed to write log: %v\n", err)
        }
}

        l.Error(text)
        os.Exit(1)

// Block holds chunks.
type Block struct {
        BaseNode
        Chunks []Chunk
}

func NewBlock(tok token.Token) *Block {
        return &Block{
                BaseNode: NewBaseNode(tok),
                Chunks:   make([]Chunk, 0),
        }
}

func (*Block) chunkNode() {}

func (b *Block) String() string {
        var out strings.Builder
        out.Grow(len(b.Chunks))

        for i := range b.Chunks {
                out.WriteString(b.Chunks[i].String())
        }

        return out.String()
}

func (b *Block) AllChunks() []Chunk {
        if b.Chunks == nil {
                return []Chunk{}
        }

        chunks := make([]Chunk, 0, len(b.Chunks))

        for _, chunk := range b.Chunks {
                if chunk == nil {
                        continue
                }

                if s, ok := chunk.(NodeWithChunks); ok {
                        chunks = append(chunks, s.(Chunk))
                        chunks = append(chunks, s.AllChunks()...)
                }
        }

        return chunks
}

func (b *Block) ExtractPassDirs() []*PassDir {
        passDirs := []*PassDir{}
        for _, chunk := range b.AllChunks() {
                if passDir, ok := chunk.(*PassDir); ok {
                        passDirs = append(passDirs, passDir)
                }
        }
        return passDirs
}

// ToDefaultPassDir removes all *PassDir chunks from the block and returns a
// *PassDir with the remaining chunks.
func (b *Block) ToDefaultPassDir(compName string) *PassDir {
        passDir := NewPassDir(*b.Tok(), NewStrExpr(*b.Tok(), ""))
        passDir.Block = NewBlock(*b.Tok())
        passDir.CompName = compName

        newChunks := []Chunk{}

        hasContent := false

        for _, chunk := range b.Chunks {
                if _, ok := chunk.(*PassDir); ok {
                        continue
                }

                newChunks = append(newChunks, chunk)
                if !isBlankText(chunk) {
                        hasContent = true
                }
        }

        // Blank text and @pass alone don't make a default slot, so the component
        // keeps its default pass nil.
        if !hasContent {
                return nil
        }

        passDir.Block.Chunks = newChunks

        return passDir
}

// isBlankText reports whether chunk is a text node with only whitespace.
func isBlankText(chunk Chunk) bool {
        text, ok := chunk.(*Text)
        if !ok {
                return false
        }
        return strings.TrimSpace(text.Token.Lit) == ""
}

                out.WriteString(cd.Argument.String())

                out.WriteString(cd.DefaultPass.String())

        }

        out.WriteString("@end")

        return out.String()

                chunks = append(chunks, id.ElseBlock.AllChunks()...)
        }

        for _, dir := range id.ElseifDirs {
                chunks = append(chunks, dir.AllChunks()...)
        }

        return chunks
}

func NewProgram(tok token.Token) *Program {
        return &Program{
                BaseNode:   NewBaseNode(tok),
                Components: []*CompDir{},
                Inserts:    map[string]*InsertDir{},
                Reserves:   map[string]*ReserveDir{},
                Slots:      map[string]*SlotDir{},
        }
}

func (p *Program) chunkNode() {}

func (p *Program) String() string {
        var out strings.Builder
        out.Grow(len(p.Chunks))

        for i := range p.Chunks {
                out.WriteString(p.Chunks[i].String())
        }

        return out.String()
}

func (p *Program) AllChunks() []Chunk {
        chunks := make([]Chunk, 0)
        if p.Chunks == nil {
                return []Chunk{}
        }

        for _, chunk := range p.Chunks {
                if chunk == nil {
                        continue
                }

                if s, ok := chunk.(NodeWithChunks); ok {
                        chunks = append(chunks, s.(Chunk))
                        chunks = append(chunks, s.AllChunks()...)
                }
        }

        return chunks
}

// LinkLayoutToUse adds Layout AST program to UseStmt for the current template
// and resets chunks to UseDir chunk only. Because we don't need anything else
// inside a template. Make sure inserts are added before this is called
// because they will be removed by this function.
func (p *Program) LinkLayoutToUse(layoutProg *Program) {
        p.UseDir.LayoutProg = layoutProg
        p.Chunks = []Chunk{p.UseDir}
}

func (p *Program) HasUseDir() bool {
        return p.UseDir != nil
}

func (p *Program) LinkPassBlocksToSlots(compDir *CompDir, compFileProg *Program) *fail.Error {
        if err := compFileProg.linkPassBlocksToSlots(compDir, p.AbsPath); err != nil {
                return err
        }

        compDir.CompProg = compFileProg
        return nil
}

func (p *Program) linkPassBlocksToSlots(compDir *CompDir, compFileAbsPath string) *fail.Error {
        if compDir.DefaultPass != nil {
                if err := p.linkBlockToDefaultSlot(compDir, compFileAbsPath); err != nil {
                        return err
                }
        }

        for _, passDir := range compDir.Passes {
                if err := p.linkBlockToSlot(passDir, compDir.Name.Val, compFileAbsPath); err != nil {
                        return err
                }
        }

        return nil
}

func (p *Program) linkBlockToDefaultSlot(compDir *CompDir, compFileAbsPath string) *fail.Error {
        slotDir, ok := p.Slots[""]
        if ok {
                slotDir.Block = compDir.DefaultPass.Block
                return nil
        }

        return fail.New(
                compDir.DefaultPass.Pos(),
                compFileAbsPath,
                fail.OriginLink,
                fail.ErrDefaultSlotNotDefined,
                compDir.Name.Val,
                compDir.Name.Val,
        )
}

func (p *Program) linkBlockToSlot(
        passDir *PassDir,
        compName string,
        compFileAbsPath string,
) *fail.Error {
        name := passDir.Name.Val
        slotDir, ok := p.Slots[name]
        if ok {
                slotDir.Block = passDir.Block
                return nil
        }

        return fail.New(
                passDir.Pos(),
                compFileAbsPath,
                fail.OriginLink,
                fail.ErrSlotNotDefined,
                compName,
                name,
        )
}

func CheckUnusedInserts(prog *Program, inserts map[string]*InsertDir) *fail.Error {
        for name := range inserts {
                if _, ok := prog.Reserves[name]; ok {
                        continue
                }

                pos := inserts[name].Pos()
                path := inserts[name].AbsPath
                name := inserts[name].Name.Val

                return fail.New(pos, path, fail.OriginLink, fail.ErrUnusedInsertDetected, name, name)

func FindDuplicatePasses(compDir *CompDir) (*PassDir, int) {
        counts := map[string]int{}
        firstSeen := map[string]*PassDir{}

        var maxSlot *PassDir
        var maxCount int

        for _, passDir := range compDir.Passes {
                name := passDir.Name.Val
                counts[name]++

                if firstSeen[name] == nil {
                        firstSeen[name] = passDir
                }

                if counts[name] > 1 && counts[name] > maxCount {
                        maxCount = counts[name]
                        maxSlot = passDir
                }
        }

        if maxCount == 0 {
                return nil, 0
        }

        return maxSlot, maxCount
}

// arrLenFunc returns the length of the given arr
func arrLenFunc(receiver value.Literal, _ ...value.Literal) (value.Literal, error) {
        elems := receiver.(*value.Arr).Elements
        length := len(elems)
        return &value.Int{Val: int64(length)}, nil
}

// arrJoinFunc joins the elements of the given arr with the given separator
func arrJoinFunc(receiver value.Literal, args ...value.Literal) (value.Literal, error) {
        var separator string

        if len(args) == 0 {
                separator = ","
        } else {
                str, ok := args[0].(*value.Str)
                if !ok {
                        msg := fmt.Sprintf(fail.ErrFuncFirstArgStr, value.ARR_VAL, "join")
                        return nil, errors.New(msg)
                }

                separator = str.Val
        }

        elems := receiver.(*value.Arr).Elements

        var out strings.Builder
        out.Grow(len(elems))

        for i := range elems {
                if i > 0 {
                        out.WriteString(separator)
                }

                out.WriteString(elems[i].String())
        }

        return &value.Str{Val: out.String()}, nil
}

// arrRandFunc returns a random element from the given arr
func arrRandFunc(receiver value.Literal, _ ...value.Literal) (value.Literal, error) {
        elems := receiver.(*value.Arr).Elements
        if len(elems) == 0 {
                return &value.Nil{}, nil
        }

        return elems[rand.Intn(len(elems))], nil
}

// arrReverseFunc reverses the elements of the given arr
func arrReverseFunc(receiver value.Literal, _ ...value.Literal) (value.Literal, error) {
        elems := receiver.(*value.Arr).Elements
        if len(elems) == 0 {
                return receiver, nil
        }

        reversed := slices.Clone(elems)
        slices.Reverse(reversed)

        return &value.Arr{Elements: reversed}, nil
}

// arrSliceFunc returns a slice of the given arr. The elements are copied, so
// the result never shares a backing array with the original arr.
func arrSliceFunc(receiver value.Literal, args ...value.Literal) (value.Literal, error) {
        elems := receiver.(*value.Arr).Elements

        if len(args) == 0 {
                msg := fmt.Sprintf(fail.ErrFuncMissingArg, value.ARR_VAL, "slice")
                return nil, errors.New(msg)
        }

        startFrom, ok := args[0].(*value.Int)
        if !ok {
                msg := fmt.Sprintf(fail.ErrFuncFirstArgInt, value.ARR_VAL, "slice")
                return nil, errors.New(msg)
        }

        start := max(int(startFrom.Val), 0)
        start = min(start, len(elems))

        if len(args) == 1 {
                return &value.Arr{Elements: slices.Clone(elems[start:])}, nil
        }

        endAt, ok := args[1].(*value.Int)
        if !ok {
                msg := fmt.Sprintf(fail.ErrFuncSecondArgInt, value.ARR_VAL, "slice")
                return nil, errors.New(msg)
        }

        end := int(endAt.Val)
        if end < 0 || end > len(elems) {
                end = len(elems)
        }

        if end < start {
                end = start
        }

        return &value.Arr{Elements: slices.Clone(elems[start:end])}, nil
}

// arrShuffleFunc shuffles the elements of the given arr
func arrShuffleFunc(receiver value.Literal, _ ...value.Literal) (value.Literal, error) {
        elems := receiver.(*value.Arr).Elements

        length := len(elems)
        if length == 0 {
                return receiver, nil
        }

        // Create a copy of the elements to shuffle
        shuffled := make([]value.Literal, length)
        copy(shuffled, elems)

        // Perform Fisher-Yates shuffle
        for i := length - 1; i > 0; i-- {
                j := rand.Intn(i + 1)                               // Pick a random index
                shuffled[i], shuffled[j] = shuffled[j], shuffled[i] // Swap elements
        }

        return &value.Arr{Elements: shuffled}, nil
}

// arrContainsFunc checks if the given arr contains the given element
func arrContainsFunc(receiver value.Literal, args ...value.Literal) (value.Literal, error) {
        if len(args) == 0 {
                msg := fmt.Sprintf(fail.ErrFuncMissingArg, value.ARR_VAL, "contains")
                return nil, errors.New(msg)
        }

        target := args[0]

        for _, el := range receiver.(*value.Arr).Elements {
                if elementsMatch(el, target) {
                        return TRUE, nil
                }
        }

        return FALSE, nil
}

// arrAppendFunc appends the given elements to the given arr
func arrAppendFunc(receiver value.Literal, args ...value.Literal) (value.Literal, error) {
        if len(args) == 0 {
                msg := fmt.Sprintf(fail.ErrFuncMissingArg, value.ARR_VAL, "append")
                return nil, errors.New(msg)
        }

        arr := receiver.(*value.Arr)
        newElems := make(
                []value.Literal,
                len(arr.Elements)+len(args),
        )

        copy(newElems, arr.Elements)

        for i := range args {
                newElems[len(arr.Elements)+i] = args[i]
        }

        return &value.Arr{Elements: newElems}, nil
}

// arrPrependFunc prepends the given elements to the given arr
func arrPrependFunc(receiver value.Literal, args ...value.Literal) (value.Literal, error) {
        if len(args) == 0 {
                msg := fmt.Sprintf(fail.ErrFuncMissingArg, value.ARR_VAL, "prepend")
                return nil, errors.New(msg)
        }

        arr := receiver.(*value.Arr)
        newElems := make([]value.Literal, len(arr.Elements)+len(args))

        copy(newElems, args)

        for i := range arr.Elements {
                newElems[len(args)+i] = arr.Elements[i]
        }

        return &value.Arr{Elements: newElems}, nil
}

// the evaluator values to carry scope and path to the current file.
type Context struct {
        scope   *value.Scope // current value's scope
        absPath string       // absolute path to the file being executed

        // inserts should be used inside of layouts.
        // - key is the name of the insert.
        // - value is evaluated ASTs into value.
        inserts map[string]value.Value

        // slots should be used inside component files.
        slots SlotsStore
}

func NewContext(scope *value.Scope, absPath string) *Context {
        if scope == nil {
                panic("scope should never be nil in evaluator context")
        }

        return &Context{
                scope:   scope,
                absPath: absPath,
                inserts: map[string]value.Value{},
                slots:   SlotsStore{},
        }
}

func (ctx *Context) childContext() *Context {
        return &Context{
                scope:   ctx.scope,
                absPath: ctx.absPath,
                inserts: ctx.inserts,
                slots:   ctx.slots,
        }
}

func (ctx *Context) childContextWithNewScope() *Context {
        childCtx := ctx.childContext()
        childCtx.scope = value.NewChildScope(ctx.scope)
        return childCtx
}

package evaluator

import (
        "time"

        "github.com/textwire/textwire/v6/config"
        "github.com/textwire/textwire/v6/pkg/ast"
        "github.com/textwire/textwire/v6/pkg/fail"
        "github.com/textwire/textwire/v6/pkg/value"
)

var (
        NIL      = &value.Nil{}
        TRUE     = &value.Bool{Val: true}
        FALSE    = &value.Bool{Val: false}
        BREAK    = &value.Break{}
        CONTINUE = &value.Continue{}
)

type Evaluator struct {
        customFunc     *config.Func
        usingTemplates bool
        usingUseDir    bool

        // config can be nil when Textwire is used for simple string and
        // file evaluation. If config is not nil, it means we use templates.
        config *config.Config

        // compDepth is the number of components that are rendering inside each
        // other right now. It goes up when a component starts rendering and down
        // when it finishes, so sibling components don't add up. A new Evaluator
        // is created for every render, so renders don't share this counter.
        compDepth int

        // global is the frozen global value used by scopes the evaluator creates
        // for layouts and components.
        global *value.Obj
}

// New returns an evaluator. A nil global falls back to the shared empty one.
func New(customFunc *config.Func, conf *config.Config, global *value.Obj) *Evaluator {
        return &Evaluator{
                customFunc:     customFunc,
                config:         conf,
                usingTemplates: conf != nil,
                global:         global,
        }
}

func (e *Evaluator) Eval(node ast.Node, ctx *Context) value.Value {
        if val := e.evalValue(node, ctx); val != nil {
                return val
        }

        if lit := e.evalLiteral(node, ctx); lit != nil {
                return lit
        }

        return e.newError(node, ctx, fail.ErrUnknownType, node)
}

func (e *Evaluator) evalValue(node ast.Node, ctx *Context) value.Value {
        switch node := node.(type) {
        case *ast.Program:
                return e.program(node, ctx)
        case *ast.Embedded:
                return e.embedded(node, ctx)
        case *ast.Block:
                return e.block(node, ctx)
        case *ast.ContinueDir:
                return CONTINUE
        case *ast.BreakDir:
                return BREAK
        case *ast.UseDir:
                return e.useDir(node, ctx)
        case *ast.ReserveDir:
                return e.reserveDir(node, ctx)
        case *ast.BreakifDir:
                return e.breakifDir(node, ctx)
        case *ast.CompDir:
                return e.compDir(node, ctx)
        case *ast.ContinueifDir:
                return e.continueifDir(node, ctx)
        case *ast.SlotDir:
                return e.slotDir(node, ctx)
        case *ast.PassDir:
                return NIL
        case *ast.DumpDir:
                return e.dumpDir(node, ctx)
        case *ast.InsertDir:
                return e.insertDir(node, ctx)
        case *ast.IfDir:
                return e.ifDir(node, ctx)
        case *ast.ForDir:
                return e.forDir(node, ctx)
        case *ast.EachDir:
                return e.eachDir(node, ctx)
        case *ast.Text:
                return &value.Text{Val: node.String()}
        }
        return nil
}

func (e *Evaluator) evalLiteral(node ast.Node, ctx *Context) value.Literal {
        switch node := node.(type) {
        case *ast.AssignStmt:
                return e.assignStmt(node, ctx)
        case *ast.IncStmt:
                return e.incStmt(node, ctx)
        case *ast.DecStmt:
                return e.decStmt(node, ctx)

        // Expressions
        case *ast.IdentExpr:
                return e.identExpr(node, ctx)
        case *ast.IndexExpr:
                return e.indexExpr(node, ctx)
        case *ast.DotExpr:
                return e.dotExpr(node, ctx)
        case *ast.StrExpr:
                return e.strExpr(node)
        case *ast.BoolExpr:
                return nativeBoolToBoolObj(node.Val)
        case *ast.ObjExpr:
                return e.objExpr(node, ctx)
        case *ast.ArrExpr:
                return e.arrExpr(node, ctx)
        case *ast.PrefixExpr:
                return e.prefixExpr(node, ctx)
        case *ast.TernaryExpr:
                return e.ternaryExpr(node, ctx)
        case *ast.InfixExpr:
                return e.infixExpr(node.Op, node.Left, node.Right, ctx)
        case *ast.CallExpr:
                return e.callExpr(node, ctx)
        case *ast.GlobalCallExpr:
                return e.globalCallExpr(node, ctx)
        case *ast.IntExpr:
                return &value.Int{Val: node.Val}
        case *ast.FloatExpr:
                return &value.Float{Val: node.Val}

        case *ast.NilExpr, *ast.Absent, *ast.Illegal:
                return NIL
        }
        return nil
}

func (e *Evaluator) program(prog *ast.Program, ctx *Context) value.Value {
        block := value.NewBlock(len(prog.Chunks))

        for i := range prog.Chunks {
                val := e.Eval(prog.Chunks[i], ctx)
                if isError(val) {
                        return val
                }
                block.Append(val)
        }

        return block
}

func (e *Evaluator) embedded(embeddedAst *ast.Embedded, ctx *Context) value.Value {
        embedded := value.NewEmbedded(len(embeddedAst.Segments), embeddedAst.IsRaw)

        for _, segment := range embeddedAst.Segments {
                segment := e.evalLiteral(segment, ctx)
                if isError(segment) {
                        return segment
                }
                embedded.Segments = append(embedded.Segments, segment)
        }

        return embedded
}

// escapedEmbedded wraps val like {{ val }} does, so it's printed with HTML
// escaping unless it's marked with the raw() function.
func escapedEmbedded(val value.Literal) *value.Embedded {
        embedded := value.NewEmbedded(1, false)
        embedded.Segments = append(embedded.Segments, val)
        return embedded
}

func (e *Evaluator) ifDir(ifDir *ast.IfDir, ctx *Context) value.Value {
        cond := e.evalLiteral(ifDir.Cond, ctx)
        if isError(cond) {
                return cond
        }

        ifCtx := ctx.childContext()
        if isTruthy(cond) {
                return e.Eval(ifDir.IfBlock, ifCtx)
        }

        for i := range ifDir.ElseifDirs {
                elseifNode := ifDir.ElseifDirs[i]

                cond = e.evalLiteral(elseifNode.Cond, ifCtx)
                if isError(cond) {
                        return cond
                }

                if isTruthy(cond) {
                        return e.Eval(elseifNode.Block, ifCtx)
                }
        }

        if ifDir.ElseBlock != nil {
                return e.Eval(ifDir.ElseBlock, ifCtx)
        }

        return NIL
}

func (e *Evaluator) block(astBlock *ast.Block, ctx *Context) value.Value {
        if astBlock == nil {
                return NIL
        }

        block := value.NewBlock(len(astBlock.Chunks))

        for i := range astBlock.Chunks {
                chunk := e.Eval(astBlock.Chunks[i], ctx)
                if isError(chunk) {
                        return chunk
                }

                block.Append(chunk)

                if value.HasBreak(chunk) || value.HasContinue(chunk) {
                        break
                }
        }

        return block
}

func (e *Evaluator) assignStmt(assignStmt *ast.AssignStmt, ctx *Context) value.Literal {
        right := e.evalLiteral(assignStmt.Right, ctx)
        if isError(right) {
                return right
        }

        return e.assignTo(assignStmt.Left, assignStmt.Right, right, ctx)
}

func (e *Evaluator) assignTo(
        left ast.Expression,
        rightExpr ast.Expression,
        rightVal value.Literal,
        ctx *Context,
) value.Literal {
        switch l := left.(type) {
        case *ast.IdentExpr:
                return e.assignIdent(l, rightVal, ctx)
        case *ast.IndexExpr:
                return e.assignIndexExp(l, rightExpr, rightVal, ctx)
        case *ast.DotExpr:
                return e.assignDotExp(l, rightExpr, rightVal, ctx)
        }

        return e.newError(
                left,
                ctx,
                fail.ErrNotSupportedAssign,
                assignTargetName(left),
        )
}

// assignTargetName names the type of an expression that cannot be assigned to.
func assignTargetName(expr ast.Expression) string {
        switch expr.(type) {
        case *ast.PrefixExpr:
                return "prefix expression"
        case *ast.InfixExpr:
                return "infix expression"
        }

        if valType := value.FromTokenToValueType(expr.Tok().Type); valType != value.ERR_VAL {
                return string(valType)
        }

        return "expression"
}

func (e *Evaluator) assignIdent(
        ident *ast.IdentExpr,
        rightVal value.Literal,
        ctx *Context,
) value.Literal {
        if err := ctx.scope.Set(ident.Name, rightVal); err != nil {
                return e.newError(ident, ctx, "%s", err.Error())
        }
        return NIL
}

func (e *Evaluator) assignIndexExp(
        indexExpr *ast.IndexExpr,
        rightExpr ast.Expression,
        rightVal value.Literal,
        ctx *Context,
) value.Literal {
        left := e.evalLiteral(indexExpr.Left, ctx)
        if isError(left) {
                return left
        }

        idx := e.evalLiteral(indexExpr.Index, ctx)
        if isError(idx) {
                return idx
        }

        switch l := left.(type) {
        case *value.Arr:
                return e.assignArrIndexExp(l, idx, indexExpr, rightExpr, rightVal, ctx)
        case *value.Obj:
                return e.assignObjIndexExp(l, idx, indexExpr, rightExpr, rightVal, ctx)
        }

        return e.newError(indexExpr, ctx, fail.ErrIndexNotSupported, left.Type())
}

func (e *Evaluator) assignArrIndexExp(
        arr *value.Arr,
        idx value.Literal,
        indexExpr *ast.IndexExpr,
        rightExpr ast.Expression,
        rightVal value.Literal,
        ctx *Context,
) value.Literal {
        if arr.IsFrozen() {
                return e.newError(indexExpr, ctx, fail.ErrReadOnlyValue)
        }

        // Index must be integer
        if !idx.Is(value.INT_VAL) {
                return e.newError(indexExpr, ctx, fail.ErrArrIndexInt, idx.Type())
        }

        index := idx.(*value.Int).Val

        if index < 0 || index >= int64(len(arr.Elements)) {
                return e.newError(indexExpr, ctx, fail.ErrArrIndexOutOfBound, index, len(arr.Elements))
        }

        if value.References(rightVal, arr) {
                return e.circularAssignError(indexExpr, rightExpr, ctx)
        }

        arr.Elements[index] = rightVal

        return NIL
}

func (e *Evaluator) assignObjIndexExp(
        obj *value.Obj,
        idx value.Literal,
        indexExpr *ast.IndexExpr,
        rightExpr ast.Expression,
        rightVal value.Literal,
        ctx *Context,
) value.Literal {
        if obj.IsFrozen() {
                return e.newError(indexExpr, ctx, fail.ErrReadOnlyValue)
        }

        if !idx.Is(value.STR_VAL) {
                return e.newError(indexExpr, ctx, fail.ErrObjIndexStr, idx.Type())
        }

        if value.References(rightVal, obj) {
                return e.circularAssignError(indexExpr, rightExpr, ctx)
        }

        key := resolveObjKey(obj, idx.(*value.Str).Val)
        obj.Pairs[key] = rightVal

        return NIL
}

func (e *Evaluator) assignDotExp(
        dotExp *ast.DotExpr,
        rightExpr ast.Expression,
        rightVal value.Literal,
        ctx *Context,
) value.Literal {
        // Evaluate the left side to get the value
        left := e.evalLiteral(dotExp.Left, ctx)
        if isError(left) {
                return left
        }

        // Get the key (property name)
        key := dotExp.Key.(*ast.IdentExpr).Name

        // Type assert that left is a value
        obj, ok := left.(*value.Obj)
        if !ok {
                return e.newError(dotExp, ctx, fail.ErrKeyOnNonObj, left.Type(), key)
        }

        if obj.IsFrozen() {
                return e.newError(dotExp, ctx, fail.ErrReadOnlyValue)
        }

        if value.References(rightVal, obj) {
                return e.circularAssignError(dotExp, rightExpr, ctx)
        }

        obj.Pairs[resolveObjKey(obj, key)] = rightVal

        return NIL
}

func (e *Evaluator) useDir(useDir *ast.UseDir, ctx *Context) value.Value {
        if useDir.LayoutProg == nil {
                if e.usingTemplates {
                        return e.newError(useDir, ctx, fail.ErrUseDirMissingLayout, useDir.Name.Val)
                }
                return e.newError(useDir, ctx, fail.ErrTemplateDirectives)
        }

        e.usingUseDir = true

        // Create new layout context and pass inserts to it
        layoutCtx := NewContext(value.NewScope(e.global), useDir.LayoutProg.AbsPath)

        // Make sure that layout is missing @use
        if useDir.LayoutProg.IsLayout && useDir.LayoutProg.HasUseDir() {
                return e.newError(useDir.LayoutProg.UseDir, layoutCtx, fail.ErrUseDirIsNotAllowed)
        }

        // Evaluate @inserts and map them into new context for layout
        for name, insertDir := range useDir.Inserts {
                insert := e.insertDir(insertDir, ctx)
                if isError(insert) {
                        return insert
                }
                layoutCtx.inserts[name] = insert
        }

        // Evaluate layout program with new context
        layout := e.Eval(useDir.LayoutProg, layoutCtx)
        if isError(layout) {
                return layout
        }

        return &value.Use{
                Path:   useDir.Name.Val,
                Layout: layout,
        }
}

func (e *Evaluator) reserveDir(reserveDir *ast.ReserveDir, ctx *Context) value.Value {
        if !e.usingTemplates {
                return e.newError(reserveDir, ctx, fail.ErrTemplateDirectives)
        }

        name := reserveDir.Name.Val
        insert, ok := ctx.inserts[name]
        if !ok {
                // Inserts are optional, NIL when not provided or fallback argument
                fallback := e.evalLiteral(reserveDir.Fallback, ctx)
                if isError(fallback) || fallback == NIL {
                        return fallback
                }
                return escapedEmbedded(fallback)
        }

        return &value.Reserve{
                Name:   name,
                Insert: insert,
        }
}

func (e *Evaluator) compDir(compDir *ast.CompDir, ctx *Context) value.Value {
        if !e.usingTemplates {
                return e.newError(compDir, ctx, fail.ErrTemplateDirectives)
        }

        name := compDir.Name.Val
        if compDir.CompProg == nil {
                return e.newError(compDir, ctx, fail.ErrUndefinedComponent, name)
        }

        compCtx := NewContext(value.NewScope(e.global), compDir.CompProg.AbsPath)

        if compCtx.slots[name] == nil {
                compCtx.slots[name] = map[string]value.Value{}
        }

        if err := e.evalCompDirPasses(compDir, ctx, compCtx); err != nil {
                return err
        }

        if compDir.Argument != nil {
                err := e.passCompArgsToCtx(compDir, ctx, compCtx)
                if err != nil {
                        return err
                }
        }

        maxDepth := e.maxCompDepth()
        if e.compDepth >= maxDepth {
                return e.newError(compDir, ctx, fail.ErrMaxComponentDepth, name, maxDepth)
        }

        e.compDepth++
        defer func() { e.compDepth-- }()

        content := e.Eval(compDir.CompProg, compCtx)
        if isError(content) {
                return content
        }

        return &value.Component{
                Name:    name,
                Content: content,
        }
}

// maxCompDepth returns how many components can render inside each other.
func (e *Evaluator) maxCompDepth() int {
        if e.config == nil || e.config.MaxComponentDepth < 1 {
                return config.DefaultMaxComponentDepth
        }
        return e.config.MaxComponentDepth
}

func (e *Evaluator) evalCompDirPasses(compDir *ast.CompDir, ctx, compCtx *Context) value.Value {
        for _, passDir := range compDir.Passes {
                if passDir.Cond != nil {
                        cond := e.evalLiteral(passDir.Cond, ctx)
                        if isError(cond) {
                                return cond
                        }

                        if !isTruthy(cond) {
                                continue
                        }
                }

                passBlock := e.block(passDir.Block, ctx)
                if isError(passBlock) {
                        return passBlock
                }

                compCtx.slots[compDir.Name.Val][passDir.Name.Val] = passBlock
        }

        // Save content from default @pass
        if compDir.DefaultPass != nil {
                passBlock := e.block(compDir.DefaultPass.Block, ctx)
                if isError(passBlock) {
                        return passBlock
                }
                compCtx.slots[compDir.Name.Val][""] = passBlock
        }

        return nil
}

func (e *Evaluator) passCompArgsToCtx(compDir *ast.CompDir, ctx, compCtx *Context) value.Value {
        for key, arg := range compDir.Argument.Pairs {
                obj := e.evalLiteral(arg, ctx)
                if isError(obj) {
                        return obj
                }

                if err := compCtx.scope.Set(key, obj); err != nil {
                        return e.newError(compDir, ctx, "%s", err.Error())
                }
        }
        return nil
}

func (e *Evaluator) forDir(forDir *ast.ForDir, ctx *Context) value.Value {
        forCtx := ctx.childContextWithNewScope()

        init := e.forDirInit(forDir.Init, forCtx)
        if isError(init) {
                return init
        }

        // Evaluate ElseBlock block if user's condition is false
        cond := e.evalLiteral(forDir.Cond, forCtx)
        if isError(cond) {
                return cond
        }

        if !isTruthy(cond) && forDir.ElseBlock != nil {
                return e.Eval(forDir.ElseBlock, forCtx)
        }

        blockLen := 0
        if forDir.Block != nil {
                blockLen = len(forDir.Block.Chunks)
        }

        block := value.NewBlock(blockLen)

        hasCond := hasForCond(forDir.Cond)
        parentLoop := nearestLoop(forCtx)
        index := 0

        // Loop through the block until the user's condition is false
        for {
                cond = e.evalLiteral(forDir.Cond, forCtx)
                if isError(cond) {
                        return cond
                }

                if hasCond && !isTruthy(cond) {
                        break
                }

                forCtx.scope.SetLoopVar(loopObj(index, -1, parentLoop))
                index++

                val := e.Eval(forDir.Block, forCtx)
                if isError(val) {
                        return val
                }

                broken := value.HasBreak(val)

                block.Append(stripControlDirs(val))

                if broken {
                        break
                }

                post := e.evalLiteral(forDir.Post, forCtx)
                if isError(post) {
                        return post
                }
        }

        return block
}

// hasForCond reports whether a @for directive has a real condition.
// A missing or empty condition means the loop runs until it breaks.
func hasForCond(cond ast.Expression) bool {
        if cond == nil {
                return false
        }

        _, isEmpty := cond.(*ast.Absent)

        return !isEmpty
}

func (e *Evaluator) forDirInit(init ast.Statement, ctx *Context) value.Literal {
        assign, ok := init.(*ast.AssignStmt)
        if !ok {
                return e.evalLiteral(init, ctx)
        }

        ident, ok := assign.Left.(*ast.IdentExpr)
        if !ok {
                return e.evalLiteral(init, ctx)
        }

        right := e.evalLiteral(assign.Right, ctx)
        if isError(right) {
                return right
        }

        if err := ctx.scope.SetLoopVariable(ident.Name, right); err != nil {
                return e.newError(init, ctx, "%s", err.Error())
        }

        return NIL
}

func (e *Evaluator) eachDir(eachDir *ast.EachDir, ctx *Context) value.Value {
        eachCtx := ctx.childContextWithNewScope()
        varName := eachDir.Var.Name

        arrObj := e.evalLiteral(eachDir.Arr, eachCtx)
        if isError(arrObj) {
                return arrObj
        }

        arr, ok := arrObj.(*value.Arr)
        if !ok {
                return e.newError(eachDir, eachCtx, fail.ErrEachDirWithNonArrArg, arrObj.Type())
        }

        arrElems := arr.Elements

        // Evaluate ElseBlock when array is empty
        if len(arrElems) == 0 && eachDir.ElseBlock != nil {
                return e.Eval(eachDir.ElseBlock, eachCtx)
        }

        block := value.NewBlock(len(arrElems))
        parentLoop := nearestLoop(eachCtx)

        for i := range arrElems {
                if err := eachCtx.scope.SetLoopVariable(varName, arrElems[i]); err != nil {
                        return e.newError(eachDir, eachCtx, "%s", err.Error())
                }

                eachCtx.scope.SetLoopVar(loopObj(i, len(arrElems), parentLoop))

                val := e.Eval(eachDir.Block, eachCtx)
                if isError(val) {
                        return val
                }

                broken := value.HasBreak(val)

                block.Append(stripControlDirs(val))
                if broken {
                        break
                }
        }

        return block
}

// nearestLoop returns the loop object from the closest active loop.
// It returns nil when the current context is not inside a loop.
func nearestLoop(ctx *Context) value.Literal {
        loop, ok := ctx.scope.Get("loop")
        if !ok {
                return nil
        }
        return loop
}

// loopObj builds the loop metadata for a loop iteration. A negative total
// means the total iteration count is unknown, so the "last" property is
// omitted. The parent field is only added for nested loops, so missing parent
// access follows the normal object-property behavior.
func loopObj(index, total int, parent value.Literal) map[string]value.Literal {
        pairs := map[string]value.Literal{
                "index": &value.Int{Val: int64(index)},
                "first": nativeBoolToBoolObj(index == 0),
                "iter":  &value.Int{Val: int64(index + 1)},
        }

        if total >= 0 {
                pairs["last"] = nativeBoolToBoolObj(index == total-1)
        }

        if parent != nil {
                pairs["parent"] = parent
        }

        return pairs
}

func (e *Evaluator) breakifDir(breakifDir *ast.BreakifDir, ctx *Context) value.Value {
        cond := e.evalLiteral(breakifDir.Cond, ctx)
        if isError(cond) {
                return cond
        }

        if isTruthy(cond) {
                return BREAK
        }

        return NIL
}

func (e *Evaluator) continueifDir(contifDir *ast.ContinueifDir, ctx *Context) value.Value {
        cond := e.evalLiteral(contifDir.Cond, ctx)
        if isError(cond) {
                return cond
        }

        if isTruthy(cond) {
                return CONTINUE
        }

        return NIL
}

func (e *Evaluator) slotDir(slotDir *ast.SlotDir, ctx *Context) value.Value {
        name := slotDir.Name.Val
        compName := slotDir.CompName

        // Get slot's content from the context
        content, ok := ctx.slots[compName][name]
        if !ok {
                // Slots are optional in component files since v3.1.0
                return NIL
        }

        return &value.Slot{Name: name, Content: content}
}

func (e *Evaluator) insertDir(insertDir *ast.InsertDir, ctx *Context) value.Value {
        if !e.usingTemplates {
                return e.newError(insertDir, ctx, fail.ErrTemplateDirectives)
        }

        name := insertDir.Name.Val
        if !e.usingUseDir {
                return e.newError(insertDir, ctx, fail.ErrInsertRequiresUse, name)
        }

        block := e.combineInsertContent(insertDir, ctx)
        if isError(block) {
                return block
        }

        return &value.Insert{
                Name:  name,
                Block: block,
        }
}

// combineInsertContent combines insert Argument and Block (depending what user has)
// into a single value that we can work with.
func (e *Evaluator) combineInsertContent(insertDir *ast.InsertDir, ctx *Context) value.Value {
        arg := e.evalLiteral(insertDir.Argument, ctx)
        if isError(arg) {
                return arg
        }

        if arg != NIL {
                return escapedEmbedded(arg)
        }

        if insertDir.Block == nil {
                return e.newError(insertDir, ctx, fail.ErrInsertMustHaveContent, insertDir.Name.Val)
        }

        return e.Eval(insertDir.Block, ctx)
}

func (e *Evaluator) dumpDir(dumpDir *ast.DumpDir, ctx *Context) value.Value {
        dump := value.NewDump(len(dumpDir.Args))

        for i := range dumpDir.Args {
                evaluated := e.evalLiteral(dumpDir.Args[i], ctx)
                dump.Vals = append(dump.Vals, evaluated)
        }

        return dump
}

func (e *Evaluator) identExpr(ident *ast.IdentExpr, ctx *Context) value.Literal {
        if val, ok := ctx.scope.Get(ident.Name); ok {
                return val
        }

        return e.newError(ident, ctx, fail.ErrVariableIsUndefined, ident.Name)
}

func (e *Evaluator) indexExpr(indexExp *ast.IndexExpr, ctx *Context) value.Literal {
        left := e.evalLiteral(indexExp.Left, ctx)
        if isError(left) {
                return left
        }

        idx := e.evalLiteral(indexExp.Index, ctx)
        if isError(idx) {
                return idx
        }

        switch {
        case left.Is(value.ARR_VAL) && idx.Is(value.INT_VAL):
                return e.arrIndexExp(left, idx)
        case left.Is(value.OBJ_VAL) && idx.Is(value.STR_VAL):
                return e.objKeyExp(left.(*value.Obj), idx.(*value.Str).Val)
        case left.Is(value.OBJ_VAL):
                return e.newError(indexExp, ctx, fail.ErrObjIndexStr, idx.Type())
        }

        return e.newError(indexExp, ctx, fail.ErrIndexNotSupported, left.Type())
}

func (e *Evaluator) arrIndexExp(arrObj, idx value.Literal) value.Literal {
        arr := arrObj.(*value.Arr)
        index := idx.(*value.Int).Val
        max := int64(len(arr.Elements) - 1)

        if index < 0 || index > max {
                return value.NewNil(value.NilOriginMissingIndex)
        }

        return arr.Elements[index]
}

func (e *Evaluator) objKeyExp(obj *value.Obj, key string) value.Literal {
        if pair, ok := obj.Pairs[key]; ok {
                return pair
        }

        // Capitalize the first letter of the key and try again to support
        // case insensitive key access for the first key character.
        if pair, ok := obj.Pairs[capitalizeFirst(key)]; ok {
                return pair
        }

        return value.NewNil(value.NilOriginMissingProp)
}

// resolveObjKey returns the existing key that reads resolve to, so writes
// update struct fields like Count instead of adding a new count key.
func resolveObjKey(obj *value.Obj, key string) string {
        if _, ok := obj.Pairs[key]; ok {
                return key
        }

        capitalized := capitalizeFirst(key)
        if _, ok := obj.Pairs[capitalized]; ok {
                return capitalized
        }

        return key
}

func (e *Evaluator) dotExpr(dotExp *ast.DotExpr, ctx *Context) value.Literal {
        left := e.evalLiteral(dotExp.Left, ctx)
        if isError(left) {
                return left
        }

        key := dotExp.Key.(*ast.IdentExpr)
        obj, ok := left.(*value.Obj)
        if !ok {
                return e.newError(dotExp, ctx, fail.ErrKeyOnNonObj, left.Type(), key)
        }

        return e.objKeyExp(obj, key.Name)
}

func (e *Evaluator) strExpr(strLit *ast.StrExpr) value.Literal {
        return &value.Str{Val: strLit.Val}
}

func (e *Evaluator) prefixExpr(prefixExp *ast.PrefixExpr, ctx *Context) value.Literal {
        right := e.evalLiteral(prefixExp.Right, ctx)
        if isError(right) {
                return right
        }

        switch prefixExp.Op {
        case "-":
                return e.minusPrefixOpExp(right, prefixExp, ctx)
        case "!":
                return e.bangOpExp(right, prefixExp, ctx)
        }

        return e.newError(prefixExp, ctx, fail.ErrUnknownOp, prefixExp.Op, right.Type())
}

func (e *Evaluator) ternaryExpr(ternExp *ast.TernaryExpr, ctx *Context) value.Literal {
        cond := e.evalLiteral(ternExp.Cond, ctx)
        if isError(cond) {
                return cond
        }

        if isTruthy(cond) {
                return e.evalLiteral(ternExp.IfExpr, ctx)
        }

        return e.evalLiteral(ternExp.ElseExpr, ctx)
}

func (e *Evaluator) arrExpr(arrLit *ast.ArrExpr, ctx *Context) value.Literal {
        elems := e.evalExpressions(arrLit.Elements, ctx)
        if len(elems) == 1 && isError(elems[0]) {
                return elems[0]
        }

        return &value.Arr{Elements: elems}
}

func (e *Evaluator) objExpr(objLit *ast.ObjExpr, ctx *Context) value.Literal {
        pairs := make(map[string]value.Literal, len(objLit.Pairs))

        for key, val := range objLit.Pairs {
                valObj := e.evalLiteral(val, ctx)
                if isError(valObj) {
                        return valObj
                }

                pairs[key] = valObj
        }

        return value.NewObj(pairs)
}

func (e *Evaluator) evalExpressions(exps []ast.Expression, ctx *Context) []value.Literal {
        evaluatedObjs := make([]value.Literal, 0, len(exps))

        for i := range exps {
                evaluated := e.evalLiteral(exps[i], ctx)
                if isError(evaluated) {
                        return []value.Literal{evaluated}
                }

                evaluatedObjs = append(evaluatedObjs, evaluated)
        }

        return evaluatedObjs
}

func (e *Evaluator) infixExpr(
        op string,
        leftNode,
        rightNode ast.Expression,
        ctx *Context,
) value.Literal {
        left := e.evalLiteral(leftNode, ctx)
        if isError(left) {
                return left
        }

        if obj, ok := e.shortCircuit(left, op); ok {
                return obj
        }

        right := e.evalLiteral(rightNode, ctx)
        if isError(right) {
                return right
        }

        if op == "&&" || op == "||" {
                return e.logicalExp(op, right, left, leftNode, ctx)
        }

        return e.operatorExp(op, left, right, leftNode, ctx)
}

// Short-circuit evaluation for logical operators to prevent
// checking conditions if the left side is false.
func (e *Evaluator) shortCircuit(left value.Literal, op string) (value.Literal, bool) {
        if op == "&&" && !isTruthy(left) {
                return FALSE, true
        }
        if op == "||" && isTruthy(left) {
                return TRUE, true
        }

        return nil, false
}

func (e *Evaluator) incStmt(incStmt *ast.IncStmt, ctx *Context) value.Literal {
        left := e.evalLiteral(incStmt.Left, ctx)
        if isError(left) {
                return left
        }

        switch l := left.(type) {
        case *value.Int:
                return e.assignTo(incStmt.Left, nil, &value.Int{Val: l.Val + 1}, ctx)
        case *value.Float:
                return e.assignTo(incStmt.Left, nil, &value.Float{Val: l.Val + 1}, ctx)
        }

        return e.newError(incStmt, ctx, fail.ErrIllegalTypeForInc, left.Type())
}

func (e *Evaluator) decStmt(decInc *ast.DecStmt, ctx *Context) value.Literal {
        left := e.evalLiteral(decInc.Left, ctx)
        if isError(left) {
                return left
        }

        switch l := left.(type) {
        case *value.Int:
                return e.assignTo(decInc.Left, nil, &value.Int{Val: l.Val - 1}, ctx)
        case *value.Float:
                return e.assignTo(decInc.Left, nil, &value.Float{Val: l.Val - 1}, ctx)
        }

        return e.newError(decInc, ctx, fail.ErrIllegalTypeForDec, left.Type())
}

func (e *Evaluator) callExpr(callExp *ast.CallExpr, ctx *Context) value.Literal {
        receiver := e.evalLiteral(callExp.Receiver, ctx)
        funcName := callExp.Function.Name
        if isError(receiver) {
                return receiver
        }

        receiverType := receiver.Type()
        typeFuncs, ok := functions[receiverType]
        if !ok {
                return e.newError(callExp, ctx, fail.ErrFuncNotDefined, receiverType, funcName)
        }

        args := e.evalExpressions(callExp.Arguments, ctx)
        if len(args) == 1 && isError(args[0]) {
                return args[0]
        }

        buitin, ok := typeFuncs[callExp.Function.Name]
        if ok {
                result, err := buitin.Fn(receiver, args...)
                if err != nil {
                        return e.newError(callExp, ctx, "%s", err.Error())
                }
                return result
        }

        if hasCustomFunc(e.customFunc, receiverType, funcName) {
                nativeArgs := e.valuesToNativeType(args)

                switch r := receiver.(type) {
                case *value.Str:
                        fun := e.customFunc.Str[funcName]
                        res := fun(r.String(), nativeArgs...)
                        return value.NativeToValue(res)
                case *value.Arr:
                        fun := e.customFunc.Arr[funcName]
                        nativeElems := e.valuesToNativeType(r.Elements)
                        res := fun(nativeElems, nativeArgs...)
                        return value.NativeToValue(res)
                case *value.Int:
                        fun := e.customFunc.Int[funcName]
                        res := fun(int(r.Val), nativeArgs...)
                        return value.NativeToValue(res)
                case *value.Float:
                        fun := e.customFunc.Float[funcName]
                        res := fun(r.Val, nativeArgs...)
                        return value.NativeToValue(res)
                case *value.Bool:
                        fun := e.customFunc.Bool[funcName]
                        res := fun(r.Val, nativeArgs...)
                        return value.NativeToValue(res)
                case *value.Obj:
                        fun := e.customFunc.Obj[funcName]
                        firstArg := r.Native()
                        res := fun(firstArg.(map[string]any), nativeArgs...)
                        return value.NativeToValue(res)
                }
        }

        return e.newError(callExp, ctx, fail.ErrFuncNotDefined, receiver.Type(), callExp.Function.Name)
}

func (e *Evaluator) globalCallExpr(globalCallExp *ast.GlobalCallExpr, ctx *Context) value.Literal {
        switch globalCallExp.Name {
        case "defined":
                return e.globalFuncDefined(globalCallExp, ctx)
        case "hasValue":
                return e.globalFuncHasValue(globalCallExp, ctx)
        case "formatDate":
                return e.globalFuncFormatDate(globalCallExp, ctx)
        case "raw":
                return e.globalFuncRaw(globalCallExp, ctx)
        default:
                return e.newError(
                        globalCallExp,
                        ctx,
                        fail.ErrGlobalFuncMissing,
                        globalCallExp.Name,
                )
        }
}

func (e *Evaluator) globalFuncDefined(
        globalCallExp *ast.GlobalCallExpr,
        ctx *Context,
) value.Literal {
        for i := range globalCallExp.Arguments {
                evaluated := e.evalLiteral(globalCallExp.Arguments[i], ctx)
                if isUndefinedError(evaluated) {
                        return FALSE
                }

                if isError(evaluated) {
                        return evaluated
                }

                if isMissingNil(evaluated) {
                        return FALSE
                }
        }

        return TRUE
}

func (e *Evaluator) globalFuncHasValue(
        globalCallExp *ast.GlobalCallExpr,
        ctx *Context,
) value.Literal {
        // Check if all arguments are defined first
        areDefined := e.globalFuncDefined(globalCallExp, ctx)
        if isError(areDefined) {
                return areDefined
        }

        if areDefined == FALSE {
                return FALSE
        }

        // At this point, all variables are defined.
        // Checking if they have nullable values.
        for _, exp := range globalCallExp.Arguments {
                arg := e.evalLiteral(exp, ctx)
                if isError(arg) {
                        return arg
                }

                if !isTruthy(arg) {
                        return FALSE
                }
        }

        return TRUE
}

func (e *Evaluator) globalFuncFormatDate(
        globalCallExpr *ast.GlobalCallExpr,
        ctx *Context,
) value.Literal {
        // Note: We already know that we have 2 arguments since parser ensures that

        date, err := e.globalCallExpectStrArg(globalCallExpr, ctx, 0)
        if err != nil {
                return err
        }

        layout, err := e.globalCallExpectStrArg(globalCallExpr, ctx, 1)
        if err != nil {
                return err
        }

        if date.Val == "" {
                return &value.Str{Val: ""}
        }

        parseLayout := getDateTimeLayout(date.Val)
        if parseLayout == "" {
                return e.newError(globalCallExpr, ctx, fail.ErrFormatDateWrongDate, date.Val)
        }

        newFormat, parseErr := time.Parse(parseLayout, date.Val)
        if parseErr != nil {
                return e.newError(globalCallExpr, ctx, fail.ErrFormatDateParseErr, date.Val, layout.Val)
        }

        return &value.Str{Val: newFormat.Format(layout.Val)}
}

// globalFuncRaw marks the strings in its argument as raw, so they're
// printed without HTML escaping. See value.Raw.
func (e *Evaluator) globalFuncRaw(
        globalCallExpr *ast.GlobalCallExpr,
        ctx *Context,
) value.Literal {
        // We already know that we have 1 argument since parser ensures that
        arg := e.evalLiteral(globalCallExpr.Arguments[0], ctx)
        if isError(arg) {
                return arg
        }

        return value.Raw(arg)
}

func (e *Evaluator) valuesToNativeType(args []value.Literal) []any {
        vals := make([]any, len(args))
        for i := range args {
                vals[i] = args[i].Native()
        }

        return vals
}

func (e *Evaluator) operatorExp(
        op string,
        left,
        right value.Literal,
        leftNode ast.Node,
        ctx *Context,
) value.Literal {
        if op == "==" || op == "!=" {
                return e.comparrisonInfixExp(op, right, left, leftNode, ctx)
        }

        switch l := left.(type) {
        case *value.Int:
                return e.intInfixExp(op, right, l, leftNode, ctx)
        case *value.Float:
                return e.floatInfixExp(op, right, l, leftNode, ctx)
        case *value.Str:
                return e.stringInfixExp(op, right, l, leftNode, ctx)
        }

        return e.newError(leftNode, ctx, fail.ErrCannotUseOperator, op, left.Type(), op, right.Type())
}

func (e *Evaluator) logicalExp(
        op string,
        right,
        left value.Literal,
        leftNode ast.Node,
        ctx *Context,
) value.Literal {
        switch op {
        case "&&":
                return &value.Bool{Val: isTruthy(left) && isTruthy(right)}
        case "||":
                return &value.Bool{Val: isTruthy(left) || isTruthy(right)}
        }

        return e.newError(leftNode, ctx, fail.ErrCannotUseOperator, op, left.Type(), op, right.Type())
}

func (e *Evaluator) intInfixExp(
        op string,
        right value.Literal,
        l *value.Int,
        leftNode ast.Node,
        ctx *Context,
) value.Literal {
        r, ok := right.(*value.Int)
        if !ok {
                return e.newError(leftNode, ctx, fail.ErrCannotUseOperator, op, l.Type(), op, right.Type())
        }

        switch op {
        case "+":
                return &value.Int{Val: l.Val + r.Val}
        case "-":
                return &value.Int{Val: l.Val - r.Val}
        case "*":
                return &value.Int{Val: l.Val * r.Val}
        case "/":
                if r.Val == 0 {
                        return e.newError(leftNode, ctx, fail.ErrDivisionByZero)
                }
                return &value.Int{Val: l.Val / r.Val}
        case "%":
                if r.Val == 0 {
                        return e.newError(leftNode, ctx, fail.ErrDivisionByZero)
                }
                return &value.Int{Val: l.Val % r.Val}
        case ">":
                return nativeBoolToBoolObj(l.Val > r.Val)
        case "<":
                return nativeBoolToBoolObj(l.Val < r.Val)
        case ">=":
                return nativeBoolToBoolObj(l.Val >= r.Val)
        case "<=":
                return nativeBoolToBoolObj(l.Val <= r.Val)
        }

        return e.newError(leftNode, ctx, fail.ErrCannotUseOperator, op, l.Type(), op, right.Type())
}

func (e *Evaluator) comparrisonInfixExp(
        op string, // == or !=
        right,
        left value.Literal,
        leftNode ast.Node,
        ctx *Context,
) value.Literal {
        if left.Type() != right.Type() {
                return nativeBoolToBoolObj(op == "!=")
        }
        var areEqual bool

        switch l := left.(type) {
        case *value.Int:
                areEqual = l.Val == right.(*value.Int).Val
        case *value.Float:
                areEqual = l.Val == right.(*value.Float).Val
        case *value.Str:
                areEqual = l.Val == right.(*value.Str).Val
        case *value.Bool:
                areEqual = l.Val == right.(*value.Bool).Val
        case *value.Arr, *value.Obj:
                areEqual = literalsEqual(left, right)
        case *value.Nil:
                areEqual = true
        default:
                e.newError(leftNode, ctx, fail.ErrCannotUseOperator, op, l.Type(), op, right.Type())
        }

        if op == "!=" {
                areEqual = !areEqual
        }

        return nativeBoolToBoolObj(areEqual)
}

func (e *Evaluator) stringInfixExp(
        op string,
        right value.Literal,
        l *value.Str,
        leftNode ast.Node,
        ctx *Context,
) value.Literal {
        r, ok := right.(*value.Str)
        if !ok {
                return e.newError(leftNode, ctx, fail.ErrCannotUseOperator, op, l.Type(), op, right.Type())
        }

        if op == "+" {
                return &value.Str{Val: l.Val + r.Val}
        }

        return e.newError(leftNode, ctx, fail.ErrCannotUseOperator, op, l.Type(), op, right.Type())
}

func (e *Evaluator) floatInfixExp(
        op string,
        right value.Literal,
        l *value.Float,
        leftNode ast.Node,
        ctx *Context,
) value.Literal {
        r, ok := right.(*value.Float)
        if !ok {
                return e.newError(leftNode, ctx, fail.ErrCannotUseOperator, op, l.Type(), op, right.Type())
        }

        switch op {
        case "+":
                return &value.Float{Val: l.Val + r.Val}
        case "-":
                return &value.Float{Val: l.Val - r.Val}
        case "*":
                return &value.Float{Val: l.Val * r.Val}
        case "/":
                if r.Val == 0 {
                        return e.newError(leftNode, ctx, fail.ErrDivisionByZero)
                }
                return &value.Float{Val: l.Val / r.Val}
        case ">":
                return nativeBoolToBoolObj(l.Val > r.Val)
        case "<":
                return nativeBoolToBoolObj(l.Val < r.Val)
        case ">=":
                return nativeBoolToBoolObj(l.Val >= r.Val)
        case "<=":
                return nativeBoolToBoolObj(l.Val <= r.Val)
        }

        return e.newError(leftNode, ctx, fail.ErrCannotUseOperator, op, l.Type(), op, right.Type())
}

func (e *Evaluator) minusPrefixOpExp(
        right value.Literal,
        node ast.Node,
        ctx *Context,
) value.Literal {
        switch r := right.(type) {
        case *value.Int:
                return &value.Int{Val: -r.Val}
        case *value.Float:
                return &value.Float{Val: -r.Val}
        }

        return e.newError(node, ctx, fail.ErrPrefixOpIsWrong, "-", right.Type())
}

func (e *Evaluator) bangOpExp(right value.Literal, node ast.Node, ctx *Context) value.Literal {
        switch right := right.(type) {
        case *value.Bool:
                return nativeBoolToBoolObj(!right.Val)
        case *value.Nil:
                return TRUE
        }

        return e.newError(node, ctx, fail.ErrPrefixOpIsWrong, "!", right.Type())
}

func (e *Evaluator) globalCallExpectStrArg(
        call *ast.GlobalCallExpr,
        ctx *Context,
        argIdx int,
) (*value.Str, *value.Error) {
        val := e.evalLiteral(call.Arguments[argIdx], ctx)
        if isError(val) {
                return nil, val.(*value.Error)
        }

        str, ok := val.(*value.Str)
        if !ok {
                return nil, e.newError(
                        call,
                        ctx,
                        fail.ErrGlobalFuncWrongType,
                        call.Name,
                        value.STR_VAL,
                        argIdx+1,
                        val.Type(),
                )
        }

        return str, nil
}

func (e *Evaluator) newError(node ast.Node, ctx *Context, format string, a ...any) *value.Error {
        return &value.Error{
                Err:     fail.New(node.Pos(), ctx.absPath, fail.OriginEval, format, a...),
                ErrorID: format,
        }
}

func (e *Evaluator) circularAssignError(
        left ast.Expression,
        rightExpr ast.Expression,
        ctx *Context,
) value.Literal {
        return e.newError(
                left,
                ctx,
                fail.ErrCircularReference,
                exprToText(rightExpr),
                exprToText(left),
        )
}

// intAbsFunc returns the absolute value of an integer
func intAbsFunc(receiver value.Literal, _ ...value.Literal) (value.Literal, error) {
        val := receiver.(*value.Int).Val
        if val == math.MinInt64 {
                msg := fmt.Sprintf(fail.ErrFuncOutOfInt64Range, value.INT_VAL, "abs")
                return nil, errors.New(msg)
        }

        if val < 0 {
                return &value.Int{Val: -val}, nil
        }

        return receiver, nil
}

func objGetFunc(receiver value.Literal, args ...value.Literal) (value.Literal, error) {
        obj := receiver.(*value.Obj)

        if len(args) == 0 {
                msg := fmt.Sprintf(fail.ErrFuncMissingArg, value.OBJ_VAL, "get")
                return nil, errors.New(msg)
        }

        if len(args) > 1 {
                msg := fmt.Sprintf(fail.ErrFuncMaxArgs, value.OBJ_VAL, "get", 1)
                return nil, errors.New(msg)
        }

        pattern, ok := args[0].(*value.Str)
        if !ok {
                msg := fmt.Sprintf(fail.ErrFuncFirstArgStr, value.OBJ_VAL, "get")
                return nil, errors.New(msg)
        }

        patternStr := pattern.String()

        if result, ok := lookupObjPair(obj.Pairs, patternStr); ok {
                return result, nil
        }

        props := strings.Split(patternStr, ".")
        return findObjKey(props, obj.Pairs), nil
}

// lookupObjPair returns the value stored under key, falling back to the
// capitalized key so struct fields (e.g. "name" -> "Name") resolve like
// dot access does.
func lookupObjPair(pairs map[string]value.Literal, key string) (value.Literal, bool) {
        if val, ok := pairs[key]; ok {
                return val, true
        }

        if val, ok := pairs[capitalizeFirst(key)]; ok {
                return val, true
        }

        return nil, false
}

func findObjKey(props []string, pairs map[string]value.Literal) value.Literal {
        current := pairs

        for i := range props {
                result, ok := lookupObjPair(current, props[i])
                if !ok {
                        return NIL
                }

                // If this is the last key, return the value
                if i == len(props)-1 {
                        return result
                }

                // If not last key, value must be an object to continue
                if result.Type() != value.OBJ_VAL {
                        return NIL
                }

                current = result.(*value.Obj).Pairs
        }

        return NIL
}

// strTruncateFunc truncates a string to a specified length and appends an ellipsis.
func strTruncateFunc(receiver value.Literal, args ...value.Literal) (value.Literal, error) {
        // Validate that at least the limit argument is provided
        if len(args) == 0 {
                msg := fmt.Sprintf(fail.ErrFuncMissingArg, value.STR_VAL, "truncate")
                return nil, errors.New(msg)
        }

        // Validate that the first argument is an integer (the limit)
        firstArg, ok := args[0].(*value.Int)
        if !ok {
                msg := fmt.Sprintf(fail.ErrFuncFirstArgInt, value.STR_VAL, "truncate")
                return nil, errors.New(msg)
        }

        val := receiver.(*value.Str).Val

        // Handle negative limits by treating them as 0
        // This prevents slice bounds errors when limit < 0
        limit := max(int(firstArg.Val), 0)

        // If the string is already shorter than or equal to the limit, return it unchanged
        // This ensures we don't truncate strings that don't need truncation
        clusters := splitClusters(val)
        if limit >= len(clusters) {
                return &value.Str{Val: val}, nil
        }

        ellipsis := "..."

        // If a custom suffix is provided as the second argument, use it instead
        if len(args) > 1 {
                secondArg, ok := args[1].(*value.Str)
                if ok {
                        ellipsis = secondArg.Val
                } else {
                        msg := fmt.Sprintf(fail.ErrFuncSecondArgStr, value.STR_VAL, "truncate")
                        return nil, errors.New(msg)
                }
        }

        // Truncate the string at the limit and append the suffix
        return &value.Str{Val: strings.Join(clusters[:limit], "") + ellipsis}, nil
}

// strDecimalFunc returns the receiver formatted as a decimal number.
// - Integers get decimal places appended ("100" → "100.00").
// - Floats are padded to the minimum decimal places ("-0.5" → "-0.50").
// - Non-numeric strings are returned unchanged.
func strDecimalFunc(receiver value.Literal, args ...value.Literal) (value.Literal, error) {
        val := receiver.(*value.Str).Val

        separator, decimals, err := getDecimalConfig(value.STR_VAL, args...)
        if err != nil {
                return nil, err

        var format func(val, separator string, decimals int) string
        switch {
        case strIsInt(val):
                format = formatIntDecimals
        case isValidFloat(val):
                format = ensureMinFloatDecimals
        default:
                return &value.Str{Val: val}, nil
        }

        if err := checkDecimalResultSize(value.STR_VAL, val, separator, decimals); err != nil {
                return nil, err
        }

        return &value.Str{Val: format(val, separator, decimals)}, nil
}

// strAtFunc returns the character at the given index in the string

        if len(args) != 0 {
                firstArg, ok := args[0].(*value.Int)
                if !ok {
                        msg := fmt.Sprintf(fail.ErrFuncFirstArgInt, value.STR_VAL, "at")
                        return nil, errors.New(msg)

                index = int(firstArg.Val)

        val := receiver.(*value.Str).Val

        if index < 0 || index >= len(clusters) {
                return &value.Nil{}, nil
        }

        return &value.Str{Val: clusters[index]}, nil
}

// strRepeatFunc returns a string repeated n times
func strRepeatFunc(receiver value.Literal, args ...value.Literal) (value.Literal, error) {
        if len(args) == 0 {
                msg := fmt.Sprintf(fail.ErrFuncMissingArg, value.STR_VAL, "repeat")
                return nil, errors.New(msg)
        }

        firstArg, ok := args[0].(*value.Int)
        if !ok {
                msg := fmt.Sprintf(fail.ErrFuncFirstArgInt, value.STR_VAL, "repeat")
                return nil, errors.New(msg)
        }

        val := receiver.(*value.Str).Val
        count := int(firstArg.Val)

        if count <= 0 || val == "" {
                return &value.Str{Val: ""}, nil
        }

        // Reject output larger than maxRepeatLen bytes.
        if count > maxRepeatLen/len(val) {
                msg := fmt.Sprintf(fail.ErrFuncResultTooLarge, value.STR_VAL, "repeat", maxRepeatLen)
                return nil, errors.New(msg)
        }

        repeated := strings.Repeat(val, count)

        return &value.Str{Val: repeated}, nil
}

// strFormatFunc embeds values into a string. Similar to sprintf in C.
func strFormatFunc(receiver value.Literal, args ...value.Literal) (value.Literal, error) {
        if len(args) == 0 {
                msg := fmt.Sprintf(fail.ErrFuncMissingArg, value.STR_VAL, "format")
                return nil, errors.New(msg)
        }

        str := receiver.(*value.Str).Val

        var argIdx int
        var out strings.Builder
        out.Grow(len(args))

        for i := 0; i < len(str); i++ {
                isPlaceholder := str[i] == '%' && i+1 < len(str) && str[i+1] == 's'
                if !isPlaceholder {
                        out.WriteByte(str[i])
                        continue
                }

                if argIdx >= len(args) {
                        out.WriteByte(str[i])
                        continue
                }

                argVal := args[argIdx].String()
                out.WriteString(argVal)
                argIdx++
                i++
        }

        return &value.Str{Val: out.String()}, nil
}

func isTruthy(obj value.Literal) bool {
        switch obj := obj.(type) {
        case *value.Bool:
                return obj.Val
        case *value.Int:
                return obj.Val != 0
        case *value.Float:
                return obj.Val != 0.0
        case *value.Str:
                return obj.Val != ""
        case *value.Nil:
                return false
        case *value.Obj:

        case *value.Arr:

        case nil:

        }

// exprToText formats an expression the way it appears in source code, without
// the decorative parentheses used by ast.Expression.String().
func exprToText(expr ast.Expression) string {
        if expr == nil {
                return ""
        }

        switch expr := expr.(type) {
        case *ast.IdentExpr:
                return expr.Name
        case *ast.IndexExpr:
                return exprToText(expr.Left) + "[" + exprToText(expr.Index) + "]"
        case *ast.DotExpr:
                return exprToText(expr.Left) + "." + exprToText(expr.Key)
        }

        text := expr.String()
        if len(text) >= 2 && text[0] == '(' && text[len(text)-1] == ')' {
                return text[1 : len(text)-1]
        }

        return text
}

// isMissingNil reports whether value is a nil produced by a missing object
// property or array index.
func isMissingNil(obj value.Value) bool {
        nilVal, ok := obj.(*value.Nil)
        if !ok {
                return false
        }

        switch nilVal.Origin() {
        case value.NilOriginMissingProp, value.NilOriginMissingIndex:
                return true
        }

        return false
}

// literalsEqual reports whether two literals represent the same value. Nil
// origins are ignored because they describe where nil came from, not its value.
// The raw mark on strings is ignored too, it only changes how they're printed.
func literalsEqual(left, right value.Literal) bool {
        switch left := left.(type) {
        case *value.Nil:
                _, ok := right.(*value.Nil)
                return ok
        case *value.Str:
                right, ok := right.(*value.Str)
                return ok && left.Val == right.Val
        case *value.Arr:
                right, ok := right.(*value.Arr)
                return ok && arrsEqual(left, right)
        case *value.Obj:
                right, ok := right.(*value.Obj)
                return ok && objsEqual(left, right)
        }

        return reflect.DeepEqual(left, right)
}

func arrsEqual(left, right *value.Arr) bool {
        if len(left.Elements) != len(right.Elements) {
                return false
        }

        for i := range left.Elements {
                if !literalsEqual(left.Elements[i], right.Elements[i]) {
                        return false
                }
        }

        return true
}

func objsEqual(left, right *value.Obj) bool {
        if len(left.Pairs) != len(right.Pairs) {
                return false
        }

        for key, leftVal := range left.Pairs {
                rightVal, ok := right.Pairs[key]
                if !ok || !literalsEqual(leftVal, rightVal) {
                        return false
                }
        }

        return true
}

// stripControlDirs removes break/continue values from a block so a loop
// doesn't pass control statements it has already handled to its parent loop.
func stripControlDirs(obj value.Value) value.Value {
        block, ok := obj.(*value.Block)
        if !ok {
                return obj
        }

        if !block.HasBreak() && !block.HasContinue() {
                return obj
        }

        stripped := value.NewBlock(len(block.Chunks()))
        for _, chunk := range block.Chunks() {
                if value.IsControlDir(chunk) {
                        continue
                }

                stripped.Append(stripControlDirs(chunk))
        }

        return stripped
}

// getDecimalConfig extracts the separator and decimal count from args.
// Both are optional and default to "." and 2. A negative decimal count
// is clamped to 0. It returns an error if there are too many arguments
// or if an argument has the wrong type.
func getDecimalConfig(
        objType value.ValueType,
        args ...value.Literal,
) (string, int, error) {
        if len(args) > 2 {
                return "", 0, fmt.Errorf(fail.ErrFuncMaxArgs, objType, "decimal", 2)
        }

        separator := "."
        decimals := 2

        if len(args) > 0 {
                arg, ok := args[0].(*value.Str)
                if !ok {
                        return "", 0, fmt.Errorf(fail.ErrFuncFirstArgStr, objType, "decimal")
                }
                separator = arg.Val
        }

        if len(args) > 1 {
                arg, ok := args[1].(*value.Int)
                if !ok {
                        return "", 0, fmt.Errorf(fail.ErrFuncSecondArgInt, objType, "decimal")
                }
                decimals = max(int(arg.Val), 0)
        }

        return separator, decimals, nil
}

// checkDecimalResultSize returns an error if adding decimals to val would make
// the result longer than maxRepeatLen bytes. Nothing is added when decimals is
// not positive, so that case always passes.
func checkDecimalResultSize(
        objType value.ValueType,
        val, separator string,
        decimals int,
) error {
        if decimals <= 0 {
                return nil
        }

        // Bytes left for decimal digits after the value and separator.
        room := maxRepeatLen - len(val) - len(separator)
        if decimals > room {
                return fmt.Errorf(fail.ErrFuncResultTooLarge, objType, "decimal", maxRepeatLen)
        }

        return nil
}

// formatIntDecimals appends decimal places to an integer string.
func formatIntDecimals(val, separator string, decimals int) string {
        if decimals == 0 {
                return val
        }
        return val + separator + strings.Repeat("0", decimals)
}

// ensureMinFloatDecimals ensures a float string has at least the requested
// decimal places.
// - Only adds decimals, never removes.
// - Only works with floats with dot separator.
// - Changes separator to the one you pass.
func ensureMinFloatDecimals(val, separator string, decimals int) string {
        intPart, fracPart, found := strings.Cut(val, ".")
        if !found {
                return val
        }

        missing := decimals - len(fracPart)
        if missing > 0 {
                fracPart += strings.Repeat("0", missing)
        }

        return intPart + separator + fracPart
}

// isValidFloat checks if string represents a valid float, e.g. "1.5",
// "-0.5", "1.". Special values like "nan" or "inf" and exponent notation
// are not valid.
func isValidFloat(s string) bool {
        intPart, fracPart, found := strings.Cut(trimSign(s), ".")
        if !found || intPart+fracPart == "" {
                return false
        }

        return isDigits(intPart) && isDigits(fracPart)
}

func strIsInt(s string) bool {
        digits := trimSign(s)
        return digits != "" && isDigits(digits)
}

// trimSign removes one leading '+' or '-' sign from s.
func trimSign(s string) string {
        if s == "" {
                return s
        }

        switch s[0] {
        case '+', '-':
                return s[1:]
        }

        return s
}

// isDigits reports whether s contains only ASCII digits.
// Empty string is valid.
func isDigits(str string) bool {
        for _, char := range str {
                if char < '0' || char > '9' {
                        return false
                }
        }
        return true
}

// capitalizeFirst returns the string with its first character uppercased using
// a stack buffer to avoid heap allocations. Returns the original string
// unchanged if the first character is not a lowercase ASCII letter.
func capitalizeFirst(s string) string {
        if len(s) == 0 {
                return s
        }

        first := s[0]
        if first < 'a' || first > 'z' {
                return s
        }

        var buf [64]byte
        if len(s) > 64 {
                return string(first-32) + s[1:]
        }

        buf[0] = first - 32
        copy(buf[1:], s[1:])

        return string(buf[:len(s)])
}

const zeroWidthJoiner = '\u200D'

// isRegionalIndicator reports whether r is one of the regional indicator
// symbols used to build flag emoji.
func isRegionalIndicator(r rune) bool {
        return r >= 0x1F1E6 && r <= 0x1F1FF
}

// isClusterExtender reports whether r continues the cluster of the rune before it.
func isClusterExtender(r rune) bool {
        skinToneMods := r >= 0x1F3FB && r <= 0x1F3FF
        return unicode.IsMark(r) || skinToneMods || r == zeroWidthJoiner
}

// splitClusters splits s into approximate user-perceived characters: each
// base rune together with any combining marks, skin tone modifiers or ZWJ
// sequence parts that follow it. Flags built from two regional indicators
// stay together too. The returned strings are substrings of s, so no
// per-cluster copying occurs.
func splitClusters(s string) []string {
        var clusters []string

        start := 0           // byte offset where the current cluster begins
        afterZWJ := false    // previous rune was a zero width joiner
        hasOpenFlag := false // current cluster ends in a regional indicator still waiting for its pair

        for i, r := range s {
                isRI := isRegionalIndicator(r)
                isExtender := isClusterExtender(r)

                continuesCluster := afterZWJ || isExtender || (isRI && hasOpenFlag)

                // Save the finished cluster when the new one begins
                if i > 0 && !continuesCluster {
                        clusters = append(clusters, s[start:i])
                        start = i
                }

                switch {
                case isRI:
                        // First RI opens a flag, the second one completes it.
                        hasOpenFlag = !hasOpenFlag
                case isExtender:
                        // Extenders don't change whether a flag is pending.
                default:
                        hasOpenFlag = false
                }

                afterZWJ = r == zeroWidthJoiner
        }

        if start < len(s) {
                clusters = append(clusters, s[start:])
        }

        return clusters
}

// getDateTimeLayout takes string date and returns Go's layout format.
// Go's time.Time layouts: https://pkg.go.dev/time#pkg-constants
func getDateTimeLayout(date string) string {
        switch {
        case len(date) == 8 && date[2] == ':' && date[5] == ':':
                return "15:04:05"
        case len(date) == 19 && date[4] == '-' && date[7] == '-' && date[10] == ' ' && date[13] == ':' && date[16] == ':':
                return "2006-01-02 15:04:05"
        case len(date) == 10 && date[4] == '-' && date[7] == '-':
                return "2006-01-02"
        }
        return ""
}

// floatToInt wraps a float as a value.Int, failing when it is outside the int64 range
func floatToInt(val float64, method string) (value.Literal, error) {
        if !floatInInt64Range(val) {
                msg := fmt.Sprintf(fail.ErrFuncOutOfInt64Range, value.FLOAT_VAL, method)
                return nil, errors.New(msg)
        }

        return &value.Int{Val: int64(val)}, nil
}

// floatInInt64Range reports whether val can be converted to int64 without overflowing
func floatInInt64Range(val float64) bool {
        return val >= math.MinInt64 && val < -math.MinInt64
}

// elementsMatch compares two literals, using deep equality for
// composite types (objects and arrays) and native equality otherwise.
func elementsMatch(a, b value.Literal) bool {
        if isComposite(a, b) {
                return literalsEqual(a, b)
        }

        return a.Native() == b.Native()
}

// isComposite reports whether both literals are objects or both are arrays.
func isComposite(a, b value.Literal) bool {
        if a.Type() != b.Type() {
                return false
        }

        return a.Type() == value.OBJ_VAL || a.Type() == value.ARR_VAL
}

// Error is the main error type for Textwire that contains all the necessary
// information about the error like the line number, file path, etc.
type Error struct {
        pos      *position.Pos
        origin   ErrOrigin
        filepath string
        message  string
}

// New creates a new Error instance of Error
func New(pos *position.Pos, filepath string, origin ErrOrigin, msg string, args ...any) *Error {
        if pos == nil {
                pos = &position.Pos{}
        }

        return &Error{
                pos:      pos,
                origin:   origin,
                filepath: filepath,
                message:  fmt.Sprintf(msg, args...),
        }
}

func (e *Error) Filepath() string {
        return e.filepath
}

func (e *Error) Origin() ErrOrigin {
        return e.origin
}

func (e *Error) Pos() *position.Pos {
        return e.pos
}

func (e *Error) Message() string {
        return e.message
}

// Meta returns the error meta information like the file path and line number
func (e *Error) Meta() string {
        var path string
        if e.filepath != "" {
                path = fmt.Sprintf(" in %s", e.filepath)
        }
        return fmt.Sprintf("Textwire ERROR%s:%d", path, e.pos.Line())
}

// String returns the full error message with all the details
func (e *Error) String() string {
        return fmt.Sprintf("[%s]: %s", e.Meta(), e.Message())
}

// FatalOnError calls log.Fatal if the error message is not empty
func (e *Error) FatalOnError() {
        if e == nil {
                return
        }

        log.Fatal(e.String())
}

// PanicOnError panics if the error message is not empty
func (e *Error) PanicOnError() {
        if e == nil {
                return
        }

        panic(e.String())
}

// PrintOnError prints the error message to the standard output
// when the error message is not empty
func (e *Error) PrintOnError() {
        if e == nil {
                return
        }

        log.Println(e.String())
}

        return errors.New(e.String())
}

        // or "home". This field can be empty when we evaluate a file outside
        // of using templating system.
        Name string

        // Rel path to the source file that starts from the root of user's project.
        // When using config.TemplateFS, relative path will exclude
        // config.TemplateDir from it to use embeded paths properly.
        Rel string

        // Abs is the absolute path to the source file starting with `/`.
        Abs string

        ModTime time.Time

        // source file extension and location to root of templates.
        config *config.Config
}

        return &SourceFile{
                Name:   strings.Trim(name, "/"),
                Rel:    trimRelPath(rel),
                Abs:    abs,
                config: c,
        }
}

// Content returns the raw content of the file as a string.
func (f *SourceFile) Content() (string, error) {
        var content []byte
        var err error

        if f.config.UsesFS() {
                content, err = fs.ReadFile(f.config.TemplateFS, f.Rel)
        } else {
                content, err = os.ReadFile(f.Abs)
        }

        if err != nil && err != io.EOF {
                return "", err
        }

        return string(content), nil
}

// ReplacePathAlias replaces Textwire alias `~` with needed file prefix.
func ReplacePathAlias(s string, shortenTo PathAlias) string {
        if s == "" {
                return ""
        }

        if s[0] == '~' {
                s = string(shortenTo) + s[1:]
        }

        return s

// joinPaths safely joins 2 paths together treating slashes correctly.
func JoinPaths(path1, path2 string) string {
        return strings.TrimRight(path1, "/") + "/" + strings.TrimLeft(path2, "/")
}

// NameToRelPath turns component and use statement names to relative path
// e.g. layouts/main will be converted to templates/layouts/main.tw
// e.g. components/book will be converted to templates/components/book.tw
func NameToRelPath(name, templDir, ext string) string {
        return JoinPaths(templDir, appendFileExt(name, ext))
}

func ToFullPath(relPath string) (string, error) {
        absPath, err := filepath.Abs(relPath)
        if err != nil {
                return "", err
        }

        return absPath, nil
}

// appendFileExt adds Textwire file extension to the end of the file if needed.
// It will ignore adding if extension already exist.
func appendFileExt(path, ext string) string {
        if path == "" || strings.HasSuffix(path, ext) {
                return path
        }
        return path + ext
}

// trimRelPath removes leading / and ./
func trimRelPath(relPath string) string {
        // Trim ./ from the beginning
        if len(relPath) > 1 && relPath[0] == '.' && relPath[1] == '/' {
                relPath = relPath[2:]
        }

        return strings.TrimLeft(relPath, "/")
}

import (
        "bytes"
        "strings"

var simpleTokens = map[byte]token.TokenType{
        '*': token.MUL,
        '?': token.QUESTION,
        '/': token.DIV,
        '%': token.MOD,
        ',': token.COMMA,
        '[': token.LBRACKET,
        ']': token.RBRACKET,
        '.': token.DOT,
        ';': token.SEMI,
        ':': token.COLON,
}

type Lexer struct {
        // input is the input string to be tokenized.
        input string

        // pos is the current character position in the input.
        pos int

        // readPos is the next character position in the input.
        readPos int

        // char is the current byte character in the input.
        char byte

        // col is the current column index on the line.
        col uint

        // prevCol is the previous column index on the line.
        prevCol uint

        // startCol is the start column index on the line.
        startCol uint

        // shouldResetCol determines if we should reset the column index to 0.
        shouldResetCol bool

        // line is the current index on the line.
        line uint

        // prevLine is the line index of the previous character.
        // NOT THE PREVIOUS LINE, BUT THE LINE OF THE PREVIOUS CHARACTER.
        prevLine uint

        // startLine is the start index on the line.
        startLine uint

        // isText determines if current character is in text or Textwire.
        isText bool

        // isDirective determines if current character is a part of directive.
        isDirective bool

        // We increment it when we find "(" and decrement when we find ")".
        // It helps to determine if we are lexing a directive.
        countDirectiveParentheses int

        // If this is 0 and we find "}}" then it's the closing token.
        // We increment it when we find "{" and decrement when we find "}".
        // It helps to determine if we are in text or Textwire.
        countCurlyBraces int

        // skippedOffsets holds the byte offsets in input that are left out of text
        // token literals.
        // `int` is the index of the byte that needs to be skipped.
        skippedOffsets map[int]struct{}
}

func New(input string) *Lexer {
        l := newLexer(input)
        l.skippedOffsets = skippedOffsets(input)

        return l
}

func newLexer(input string) *Lexer {
        l := &Lexer{
                input:       input,
                isText:      true,
                isDirective: false,
        }

        // set l.char to the first character
        l.readChar()

        return l
}

func (l *Lexer) Next() token.Token {
        if !l.isText {
                l.skipWhitespace()
        }

        if l.atEnd() {
                l.tokenBegins()
                return l.newToken(token.EOF, "")
        }

        if l.startsWith('{', '{', '-', '-') {
                if !l.skipComment() {
                        l.tokenBegins()
                        return l.newToken(token.UNCLOSED_COMMENT, "")
                }
                return l.Next()
        }

        if l.startsWith('{', '{') {
                return l.bracesToken(token.LBRACES, "{{")
        }

        if l.startsWith('{', '!', '!') {
                return l.rawBracesToken(token.LBRACESRAW, "{!!")
        }

        if l.startsWith('}', '}') && l.countCurlyBraces == 0 {
                return l.bracesToken(token.RBRACES, "}}")
        }

        if l.startsWith('!', '!', '}') && l.countCurlyBraces == 0 {
                return l.rawBracesToken(token.RBRACESRAW, "!!}")
        }

        if l.isDirectiveToken() {
                return l.directiveToken()
        }

        if !l.isText {
                return l.embeddedCodeToken()
        }

        text := l.readText()
        if text == "" {
                return l.newToken(token.EMPTY, "")
        }

        return l.newToken(token.TEXT, text)
}

// atEnd reports whether the lexer has consumed all input.
func (l *Lexer) atEnd() bool {
        return l.pos >= len(l.input)
}

func (l *Lexer) startsWith(chars ...byte) bool {
        if len(chars) == 0 {
                return false
        }

        if chars[0] != l.char {
                return false
        }

        for i := 1; i < len(chars); i++ {
                if l.peek(i-1) != chars[i] {
                        return false
                }
        }

        return true
}

func (l *Lexer) bracesToken(tok token.TokenType, literal string) token.Token {
        l.isText = tok != token.LBRACES

        l.tokenBegins()
        l.readChars(2) // skip braces

        return l.newToken(tok, literal)
}

func (l *Lexer) rawBracesToken(tok token.TokenType, literal string) token.Token {
        l.isText = tok != token.LBRACESRAW

        l.tokenBegins()
        l.readChars(3) // skip braces

        return l.newToken(tok, literal)
}

func (l *Lexer) illegalToken() token.Token {
        l.tokenBegins()

        _, size := utf8.DecodeRuneInString(l.input[l.pos:])
        tok := l.newToken(token.ILLEGAL, l.input[l.pos:l.pos+size])
        l.readChars(size)

        return tok
}

func (l *Lexer) directiveToken() token.Token {
        if l.char != '@' {
                return l.illegalToken()
        }

        tok, keyword := l.readDirective()

        if tok == token.ILLEGAL {
                return l.illegalToken()
        }

        l.isDirective = l.char == '(' || l.nextNonSpaceIs('(')
        l.isText = !l.isDirective

        return l.newToken(tok, keyword)
}

func (l *Lexer) embeddedCodeToken() token.Token {
        // check simple tokens first
        if tok, ok := simpleTokens[l.char]; ok {
                c := l.char
                l.tokenBegins()
                l.readChar() // skip the l.char
                return l.newToken(tok, string(c))
        }

        switch l.char {
        case '{':
                return l.leftBraceToken()
        case '}':
                return l.rightBraceToken()
        case '(':
                return l.leftParenthesesToken()
        case ')':
                return l.rightParenthesesToken()
        case '"', '\'':
                literal, terminated := l.readString()
                if !terminated {
                        return l.newToken(token.ILLEGAL, literal)
                }
                return l.newToken(token.STR, literal)
        case '<':
                return l.operatorToken('=', token.LTHAN_EQ, token.LTHAN, "<=", "<")
        case '>':
                return l.operatorToken('=', token.GTHAN_EQ, token.GTHAN, ">=", ">")
        case '!':
                return l.operatorToken('=', token.NOT_EQ, token.NOT, "!=", "!")
        case '-':
                return l.operatorToken('-', token.DEC, token.SUB, "--", "-")
        case '&':
                if l.peek(0) == '&' {
                        return l.twoCharToken(token.AND, "&&")
                }
                return l.illegalToken()
        case '|':
                if l.peek(0) == '|' {
                        return l.twoCharToken(token.OR, "||")
                }
                return l.illegalToken()
        case '+':
                return l.operatorToken('+', token.INC, token.ADD, "++", "+")
        case '=':
                return l.operatorToken('=', token.EQ, token.ASSIGN, "==", "=")
        }

        if isIdent(l.char) {
                ident := l.readIdentifier()
                return l.newToken(token.LookupIdent(ident), ident)
        }

        if isNumber(l.char) {
                return l.numberToken()
        }

        return l.illegalToken()
}

func (l *Lexer) numberToken() token.Token {
        num, isInt := l.readNumber()

        if isInt {
                return l.newToken(token.INT, num)
        }

        return l.newToken(token.FLOAT, num)
}

func (l *Lexer) leftBraceToken() token.Token {
        l.countCurlyBraces += 1
        l.tokenBegins()
        l.readChar() // skip "{"

        return l.newToken(token.LBRACE, "{")
}

func (l *Lexer) rightBraceToken() token.Token {
        l.countCurlyBraces -= 1
        l.tokenBegins()
        l.readChar() // skip "}"

        return l.newToken(token.RBRACE, "}")
}

func (l *Lexer) leftParenthesesToken() token.Token {
        if l.isDirective {
                l.countDirectiveParentheses += 1
        }

        l.tokenBegins()
        l.readChar() // skip "("

        return l.newToken(token.LPAREN, "(")
}

func (l *Lexer) rightParenthesesToken() token.Token {
        if l.isDirective {
                l.countDirectiveParentheses -= 1
        }

        if l.isDirective && l.countDirectiveParentheses == 0 {
                l.isDirective = false
                l.isText = true
        }

        l.tokenBegins()
        l.readChar() // skip ")"

        return l.newToken(token.RPAREN, ")")
}

func (l *Lexer) twoCharToken(tokType token.TokenType, literal string) token.Token {
        l.tokenBegins()
        l.readChars(2)
        return l.newToken(tokType, literal)
}

// operatorToken handles the common pattern of checking for a two-char operator
// or falling back to single-char.
func (l *Lexer) operatorToken(
        second byte,
        twoCharTok, singleTok token.TokenType,
        twoCharLit, singleLit string,
) token.Token {
        if l.peek(0) == second {
                return l.twoCharToken(twoCharTok, twoCharLit)
        }

        l.tokenBegins()
        l.readChar()

        return l.newToken(singleTok, singleLit)
}

func (l *Lexer) newToken(tokType token.TokenType, literal string) token.Token {
        endCol := l.col
        endLine := l.line

        // We need to set the end column and line to the values of the previous
        // character because we already read the last character and incremented
        // the column index.
        // For EOF, ILLEGAL and UNCLOSED_COMMENT we don't need to decrement the column index.
        if tokType != token.EOF && tokType != token.ILLEGAL && tokType != token.UNCLOSED_COMMENT {
                endCol = l.prevCol
                endLine = l.prevLine
        }

        pos := &position.Pos{
                StartCol:  l.startCol,
                EndCol:    endCol,
                StartLine: l.startLine,
                EndLine:   endLine,
        }

        return token.Token{
                Type: tokType,
                Lit:  literal,
                Pos:  pos,
        }
}

func (l *Lexer) readIdentifier() string {
        pos := l.pos

        l.tokenBegins()

        for isIdent(l.char) || isNumber(l.char) {
                l.readChar()
        }

        return l.input[pos:l.pos]
}

func (l *Lexer) readDirective() (token.TokenType, string) {
        var keyword strings.Builder
        tok := token.ILLEGAL

        l.tokenBegins()

        for isLetterWord(l.char) && (l.hasIfVariant(tok) || tok == token.ILLEGAL) {
                keyword.WriteByte(l.char)
                tok = token.LookupDirective(keyword.String())
                l.readChar()
        }

        return tok, keyword.String()
}

func (l *Lexer) hasDirectivePrefix() bool {
        if l.char != '@' {
                return false
        }

        pos := l.pos
        longestDir := token.LongestDirective()

        for i := 1; i <= longestDir; i++ {
                if pos+i > len(l.input) {
                        return false
                }

                if token.LookupDirective(l.input[pos:pos+i]) != token.ILLEGAL {
                        return true
                }
        }

        return false
}

func (l *Lexer) isDirectiveToken() bool {
        return l.prevChar() != '\\' && l.hasDirectivePrefix()
}

func (l *Lexer) isEscapedDirective() bool {
        return l.prevChar() == '\\' && l.hasDirectivePrefix()
}

func (l *Lexer) hasIfVariant(tok token.TokenType) bool {
        // Tokens that can be extended with "if" variants, like @breakif
        optionalIfTokens := map[token.TokenType]struct{}{
                token.ELSE:     {},
                token.BREAK:    {},
                token.CONTINUE: {},
                token.PASS:     {},
        }

        if _, ok := optionalIfTokens[tok]; !ok {
                return false
        }

        return l.startsWith('i', 'f')
}

// readString reads a quoted string literal starting at the opening quote.
// It returns the string with escape sequences processed, and whether the
// closing quote was found before end of input.
func (l *Lexer) readString() (string, bool) {
        quote := l.char

        l.tokenBegins()
        l.readChar() // skip the opening quote

        start := l.pos
        terminated := l.scanToClosingQuote(quote)
        raw := l.input[start:l.pos]

        if !terminated {
                return raw, false // keep the raw text for the ILLEGAL token
        }

        l.readChar() // skip the closing quote

        return handleEscapeSequences(raw, quote), true
}

// scanToClosingQuote advances until it reaches an unescaped quote or EOF.
// It leaves the lexer positioned on the closing quote and reports whether
// one was found.
func (l *Lexer) scanToClosingQuote(quote byte) bool {
        for !l.atEnd() {
                switch l.char {
                case quote:
                        return true
                case '\\':
                        l.readChar() // skip the escaped character
                        if l.atEnd() {
                                return false
                        }
                }
                l.readChar()
        }
        return false
}

// handleEscapeSequences decodes backslash escapes in a raw string literal.
// Recognized escapes are \n, \t, \r, \\ and an escaped quote. Any other
// escape, and a trailing lone backslash, is kept as written.
func handleEscapeSequences(s string, quote byte) string {
        if strings.IndexByte(s, '\\') < 0 {
                return s // nothing to decode
        }

        var out strings.Builder
        out.Grow(len(s))

        for i := 0; i < len(s); i++ {
                isEscape := s[i] == '\\' && i+1 < len(s)

                if !isEscape {
                        out.WriteByte(s[i])
                        continue
                }

                i++ // move onto the escaped character

                if decoded, ok := decodeEscape(s[i], quote); ok {
                        out.WriteByte(decoded)
                } else {
                        out.WriteByte('\\')
                        out.WriteByte(s[i])
                }
        }

        return out.String()
}

// decodeEscape returns the byte that the escape `\<c>` stands for,
// or false if the escape is not recognized.
func decodeEscape(c, quote byte) (byte, bool) {
        switch c {
        case 'n':
                return '\n', true
        case 't':
                return '\t', true
        case 'r':
                return '\r', true
        case '\\', quote:
                return c, true
        }
        return 0, false
}

func (l *Lexer) readNumber() (string, bool) {
        pos := l.pos
        isInt := true
        l.tokenBegins()

        for isNumber(l.char) {
                l.readChar()
        }

        if l.char == '.' && isNumber(l.peek(0)) {
                isInt = false
                l.readChar()

                for isNumber(l.char) {
                        l.readChar()
                }
        }

        return l.input[pos:l.pos], isInt
}

func (l *Lexer) areBracesToken() (areBraces bool, escapedBraces bool) {
        braces := l.startsWith('{', '{') || l.startsWith('{', '!', '!')
        escapedBraces = l.prevChar() == '\\' && braces

        return braces && l.prevChar() != '\\', escapedBraces
}

func (l *Lexer) tokenBegins() {
        l.startCol = l.col
        l.startLine = l.line
}

func (l *Lexer) readText() string {
        var out bytes.Buffer
        l.tokenBegins()

        for l.isText && !l.atEnd() {
                areBraces, escapedBraces := l.areBracesToken()

                if areBraces || l.isDirectiveToken() {
                        break
                }

                if escapedBraces || l.isEscapedDirective() {
                        out.Truncate(out.Len() - 1)
                }

                if _, skip := l.skippedOffsets[l.pos]; !skip {
                        out.WriteByte(l.char)
                }
                l.readChar()
        }

        return out.String()
}

func (l *Lexer) prevChar() byte {
        if l.pos > 0 {
                return l.input[l.pos-1]
        }
        return 0
}

func (l *Lexer) readChar() {
        l.prevLine = l.line

        if l.readPos >= len(l.input) {
                l.char = 0
        } else {
                l.char = l.input[l.readPos]
        }

        l.pos = l.readPos
        l.readPos += 1

        // we don't need to increment this column on lexer
        // initialization because it should be 0 when it starts
        if l.pos > 0 {
                l.prevCol = l.col

                // UTF-8 continuation bytes are part of the previous character, so
                // they must not advance the column.
                if !l.isContinuationByte() {
                        l.col += 1
                }
        }

        if l.shouldResetCol {
                l.shouldResetCol = false
                l.col = 0
                l.line += 1
        }

        l.shouldResetCol = l.char == '\n'
}

// isContinuationByte reports whether the current byte continues
// a UTF-8 encoded character.
func (l *Lexer) isContinuationByte() bool {
        return l.char&0xC0 == 0x80
}

func (l *Lexer) readChars(chars int) {
        for range chars {
                l.readChar()
        }
}

func (l *Lexer) peek(ahead int) byte {
        if l.readPos+ahead >= len(l.input) {
                return 0
        }

        return l.input[l.readPos+ahead]
}

func (l *Lexer) isWhitespace(char byte) bool {
        return char == ' ' || char == '\t' || char == '\n' || char == '\r'
}

// nextNonSpaceIs checks if the next non-space/tab character matches
// the given character without consuming any input.
func (l *Lexer) nextNonSpaceIs(char byte) bool {
        pos := l.readPos
        for pos < len(l.input) {
                c := l.input[pos]
                if c != ' ' && c != '\t' {
                        return c == char
                }
                pos++
        }
        return false
}

func (l *Lexer) skipWhitespace() {
        for l.isWhitespace(l.char) {
                l.readChar()
        }
}

// skipComment skips a comment and reports whether its closing "--}}" was found.
func (l *Lexer) skipComment() bool {
        depth := 1

        l.readChars(4) // skip "{{--"

        for !l.atEnd() && depth > 0 {
                if l.startsWith('{', '{', '-', '-') {
                        l.readChars(4) // skip "{{--"
                        depth++
                        continue
                }

                if l.startsWith('-', '-', '}', '}') {
                        l.readChars(4) // skip "--}}"
                        depth--
                        continue
                }

                l.readChar()
        }

        l.isText = true

        return depth == 0
}

type directiveSpan struct {
        start int
        end   int
}

// skippedOffsets returns the byte offsets of lines whose only non-whitespace
// content is a single directive, including the line ending. Directive bytes are
// included, but they never reach text tokens, so only the surrounding
// whitespace is observable.
func skippedOffsets(input string) map[int]struct{} {
        skipped := map[int]struct{}{}
        dirIDs := directiveIDs(input, directiveSpans(input))

        for start := 0; start < len(input); {
                end, next := lineBounds(input, start)

                if lineHasSingleDirective(input, dirIDs, start, end) {
                        markRange(skipped, start, next)
                }

                start = next
        }

        return skipped
}

// directiveIDs labels every offset with the 1-based index of the
// directive span it belongs to, or 0 when no directive covers it.
func directiveIDs(input string, spans []directiveSpan) []int {
        ids := make([]int, len(input))

        for i, span := range spans {
                for j := span.start; j < span.end && j < len(input); j++ {
                        ids[j] = i + 1
                }
        }

        return ids
}

// lineBounds returns the offset where the content of the line starting at
// start ends, not counting the line ending, and the offset where the next
// line starts.
func lineBounds(input string, start int) (contentEnd, next int) {
        i := strings.IndexByte(input[start:], '\n')
        if i < 0 {
                return len(input), len(input)
        }

        newline := start + i
        contentEnd = newline

        if newline > start && input[newline-1] == '\r' {
                contentEnd--
        }

        return contentEnd, newline + 1
}

// lineHasSingleDirective reports whether every non-space character on the
// line belongs to the same directive. Embedded code, text and a second
// directive all make the line non-standalone.
func lineHasSingleDirective(input string, dirIDs []int, start, end int) bool {
        id := 0

        for i := start; i < end; i++ {
                if isInlineSpace(input[i]) {
                        continue
                }

                switch {
                case dirIDs[i] == 0:
                        return false
                case id == 0:
                        id = dirIDs[i]
                case id != dirIDs[i]:
                        return false
                }
        }

        return id != 0
}

// directiveSpans returns the source span of every directive, including its
// parenthesized arguments.
func directiveSpans(input string) []directiveSpan {
        var spans []directiveSpan

        l := newLexer(input)
        tok := l.Next()

        for tok.Type != token.EOF {
                if !tokenIsDirective(tok) {
                        tok = l.Next()
                        continue
                }

                span, next := scanDirective(l, tok)
                spans = append(spans, span)
                tok = next
        }

        return spans
}

// scanDirective returns the span of the directive tok, including its
// parenthesized arguments, and the token after it.
func scanDirective(l *Lexer, tok token.Token) (directiveSpan, token.Token) {
        span := directiveSpan{
                start: l.pos - len(tok.Lit),
                end:   l.pos,
        }

        tok = l.Next()
        if tok.Type == token.LPAREN {
                span.end, tok = skipDirectiveArgs(l)
        }

        return span, tok
}

// skipDirectiveArgs consumes tokens up to the parenthesis that closes the
// directive arguments. It returns the end offset and the token after it.
func skipDirectiveArgs(l *Lexer) (int, token.Token) {
        depth := 1

        for {
                tok := l.Next()

                switch tok.Type {
                case token.EOF:
                        return l.pos, tok
                case token.LPAREN:
                        depth++
                case token.RPAREN:
                        depth--
                        if depth == 0 {
                                // capture the offset before l.Next() moves l.pos
                                end := l.pos
                                return end, l.Next()
                        }
                }
        }
}

// tokenIsDirective reports whether tok is a directive keyword. The type
// check keeps string literals such as "@if" from matching.
func tokenIsDirective(tok token.Token) bool {
        return tok.Type != token.ILLEGAL && token.LookupDirective(tok.Lit) == tok.Type
}

func markRange(skipped map[int]struct{}, from, to int) {
        for i := from; i < to; i++ {
                skipped[i] = struct{}{}
        }
}

func isInlineSpace(ch byte) bool {
        return ch == ' ' || ch == '\t'
}

// NodeLinker handles connecting AST nodes between each other to prepare AST
// for evaluator. It will connect @insert to @reserve, @use to layout file,
// @component to its corresponding component file, etc.
type NodeLinker struct {
        Programs  []*ast.Program
        LinkError *fail.Error // Stores the last linking error, if any
        mu        sync.RWMutex
}

func New(progs []*ast.Program) *NodeLinker {
        if progs == nil {
                progs = make([]*ast.Program, 0, 4)
        }
        return &NodeLinker{Programs: progs}
}

// LinkNodes links components and layouts to those programs that use them.
// For example, we need to add component program to @component('book'), where
// CompProg is the parsed program AST of the `book.tw` component.
func (nl *NodeLinker) LinkNodes() *fail.Error {
        nl.unlinkAll()

        for _, prog := range nl.Programs {
                if err := nl.handleLayoutLinking(prog); err != nil {
                        return err
                }
        }

        for _, prog := range nl.Programs {
                if err := nl.handleCompLinking(prog); err != nil {
                        return err
                }
        }

        return nil
}

// unlinkAll unlinks everything from AST nodes to ensure clean state.
func (nl *NodeLinker) unlinkAll() {
        for _, prog := range nl.Programs {
                // Unlink components
                for _, comp := range prog.Components {
                        comp.CompProg = nil
                }

                // Unlink layouts and inserts
                if prog.UseDir != nil {
                        prog.UseDir.LayoutProg = nil
                        prog.UseDir.Inserts = nil
                }
        }
}

// handleLayoutLinking links layout directives to template directives
func (nl *NodeLinker) handleLayoutLinking(prog *ast.Program) *fail.Error {
        if !prog.HasUseDir() {
                return nil
        }

        prog.UseDir.Inserts = prog.Inserts

        layoutName := prog.UseDir.Name.Val
        layoutProg := ast.FindProg(layoutName, nl.Programs)
        if layoutProg == nil {
                return fail.New(
                        prog.UseDir.Name.Pos(),
                        prog.AbsPath,
                        fail.OriginLink,
                        fail.ErrUseDirMissingLayout,
                        layoutName,
                )
        }

        layoutProg.IsLayout = true
        if err := ast.CheckUnusedInserts(layoutProg, prog.Inserts); err != nil {
                return err
        }

        prog.LinkLayoutToUse(layoutProg)

        return nil
}

// handleCompLinking links component directives with component files
func (nl *NodeLinker) handleCompLinking(prog *ast.Program) *fail.Error {
        for _, compDir := range prog.Components {
                compFileProg := ast.FindProg(compDir.Name.Val, nl.Programs)
                if compFileProg == nil {
                        return fail.New(
                                compDir.Pos(),
                                prog.AbsPath,
                                fail.OriginLink,
                                fail.ErrUndefinedComponent,
                                compDir.Name.Val,
                        )
                }

                if err := prog.LinkPassBlocksToSlots(compDir, compFileProg); err != nil {
                        return err
                }
        }

        return nil
}

// Lock acquires the write lock for updating Programs.
func (nl *NodeLinker) Lock() {
        nl.mu.Lock()
}

// Unlock releases the write lock.
func (nl *NodeLinker) Unlock() {
        nl.mu.Unlock()
}

// RLock acquires the read lock for accessing Programs.
func (nl *NodeLinker) RLock() {
        nl.mu.RLock()
}

// RUnlock releases the read lock.
func (nl *NodeLinker) RUnlock() {
        nl.mu.RUnlock()
}

// IsInLoop checks if given position of the cursor is inside of a loop
func IsInLoop(doc, filePath string, line, col uint) (bool, []*fail.Error) {
        l := lexer.New(doc)
        p := parser.New(l, nil)
        program := p.ParseProgram()

        if program == nil {
                return false, p.Errors()
        }

        for _, chunk := range program.AllChunks() {
                isEachLoop := chunk.Tok().Type == token.EACH
                isForLoop := chunk.Tok().Type == token.FOR

                if !isEachLoop && !isForLoop {
                        continue
                }

                loopStmt := chunk.(ast.LoopDirective)

                block := loopStmt.LoopBlock()
                if block == nil {
                        continue
                }

                pos := block.Pos()

                if IsCursorInBlock(line, col, pos) {
                        return true, p.Errors()
                }
        }

        return false, p.Errors()
}

func IsCursorInBlock(line, col uint, pos *position.Pos) bool {
        // Line outside range
        if line < pos.StartLine || line > pos.EndLine {
                return false
        }

        // For inlined loops that are written in a single line
        if line == pos.StartLine && line == pos.EndLine {
                return col >= pos.StartCol && col <= pos.EndCol
        }

        // When cursor is on the start line
        if line == pos.StartLine {
                return col >= pos.StartCol
        }

        // When cursor is on the end line
        if line == pos.EndLine {
                return col <= pos.EndCol
        }

        // In middle lines any column is valid
        return true
}

// Locale represents a language locale for metadata.
// It's a 2 letter ISO 639-1 code (e.g. "en", "es", "fr").
// Codes: https://en.wikipedia.org/wiki/List_of_ISO_639_language_codes
type Locale string

var (
        //go:embed metadata/*
        files embed.FS

        //go:embed inserts/*
        insertFiles embed.FS

        fileNamesOnce sync.Once
        fileNames     map[token.TokenType]string

        validLocales = []Locale{"en"}
)

// GetTokenMeta retrieves metadata for the given token type and locale.
func GetTokenMeta(tok token.TokenType, locale Locale) (string, error) {
        if !isValidLocale(locale) {
                return "", utils.ErrInvalidLocale(string(locale))
        }

        fileNamesOnce.Do(initFileNames)

        fileName, ok := fileNames[tok]
        if !ok {
                return "", utils.ErrNoMetadataFound(tok)
        }

        filePath := path.Join("metadata", string(locale), fileName)

        data, err := files.ReadFile(filePath)
        if err != nil {
                return "", utils.FailedToReadFile("meta", filePath, err)
        }

        return string(data), nil
}

// GetTokenInsert retrieves insert string for the given token type. This
// insert is used for autocompletion.
func GetTokenInsert(tok token.TokenType) (string, error) {
        fileNamesOnce.Do(initFileNames)

        fileName, ok := fileNames[tok]
        if !ok {
                return "", utils.ErrNoMetadataFound(tok)
        }

        filePath := path.Join("inserts", fileName)

        data, err := insertFiles.ReadFile(filePath)
        if err != nil {
                return "", utils.FailedToReadFile("insert", filePath, err)
        }

        return string(data), nil
}

func initFileNames() {
        fileNames = map[token.TokenType]string{}

        for dir, tok := range token.GetDirectives() {
                name := strings.ToLower(dir[1:])
                fileNames[tok] = name + ".md"
        }
}

func isValidLocale(locale Locale) bool {
        return slices.Contains(validLocales, locale)
}

var precedences = map[token.TokenType]int{
        token.QUESTION: TERNARY,
        token.OR:       LOGICAL_OR,
        token.AND:      LOGICAL_AND,
        token.EQ:       EQ,
        token.NOT_EQ:   EQ,
        token.LTHAN:    LESS_GREATER,
        token.GTHAN:    LESS_GREATER,
        token.LTHAN_EQ: LESS_GREATER,
        token.GTHAN_EQ: LESS_GREATER,
        token.ADD:      SUM,
        token.SUB:      SUM,
        token.DIV:      PRODUCT,
        token.MOD:      PRODUCT,
        token.MUL:      PRODUCT,
        token.LBRACKET: INDEX,
        token.DOT:      MEMBER_ACCESS,
        token.LPAREN:   CALL,
}

type Parser struct {
        l      *lexer.Lexer
        errors []*fail.Error

        file *file.SourceFile

        curToken  token.Token
        peekToken token.Token

        // Ending token can be }} or !}} for embedded
        tokenEnd token.TokenType

        // loopDepth is the number of loops we are currently parsing. It's used to
        // reject @break and @continue outside of a loop.
        loopDepth int

        prefixParseFns map[token.TokenType]prefixParseFn
        infixParseFns  map[token.TokenType]infixParseFn

        prog *ast.Program
}

func New(lexer *lexer.Lexer, f *file.SourceFile) *Parser {
        if f == nil {
                f = file.New("", "", "", nil)
        }

        p := &Parser{
                l:      lexer,
                file:   f,
                errors: []*fail.Error{},
        }

        p.nextToken() // fill curToken
        p.nextToken() // fill peekToken

        // Prefix operators
        p.prefixParseFns = map[token.TokenType]prefixParseFn{}

        p.registerPrefix(token.IDENT, p.ident)
        p.registerPrefix(token.INT, p.intExpr)
        p.registerPrefix(token.FLOAT, p.floatExpr)
        p.registerPrefix(token.STR, p.strExpr)
        p.registerPrefix(token.NIL, p.nilExpr)
        p.registerPrefix(token.TRUE, p.boolExpr)
        p.registerPrefix(token.FALSE, p.boolExpr)
        p.registerPrefix(token.SUB, p.prefixExpr)
        p.registerPrefix(token.NOT, p.prefixExpr)
        p.registerPrefix(token.LPAREN, p.groupedExpr)
        p.registerPrefix(token.LBRACKET, p.arrExpr)
        p.registerPrefix(token.LBRACE, p.objExpr)

        // Infix operators
        p.infixParseFns = map[token.TokenType]infixParseFn{}
        p.registerInfix(token.ADD, p.infixExpr)
        p.registerInfix(token.SUB, p.infixExpr)
        p.registerInfix(token.MUL, p.infixExpr)
        p.registerInfix(token.DIV, p.infixExpr)
        p.registerInfix(token.MOD, p.infixExpr)

        p.registerInfix(token.EQ, p.infixExpr)
        p.registerInfix(token.NOT_EQ, p.infixExpr)
        p.registerInfix(token.LTHAN, p.infixExpr)
        p.registerInfix(token.GTHAN, p.infixExpr)
        p.registerInfix(token.LTHAN_EQ, p.infixExpr)
        p.registerInfix(token.GTHAN_EQ, p.infixExpr)
        p.registerInfix(token.AND, p.infixExpr)
        p.registerInfix(token.OR, p.infixExpr)

        p.registerInfix(token.QUESTION, p.ternaryExpr)
        p.registerInfix(token.LBRACKET, p.indexExpr)
        p.registerInfix(token.DOT, p.dotExpr)

        return p
}

func (p *Parser) ParseProgram() *ast.Program {
        p.prog = ast.NewProgram(p.curToken)
        p.prog.AbsPath = p.file.Abs
        p.prog.Chunks = []ast.Chunk{}

        for !p.curTokenIs(token.EOF) {
                if chunk := p.chunk(); chunk != nil {
                        p.prog.Chunks = append(p.prog.Chunks, chunk)
                }
                p.nextToken()
        }

        return p.prog
}

func (p *Parser) Errors() []*fail.Error {
        return p.errors
}

func (p *Parser) HasErrors() bool {
        return len(p.errors) > 0
}

func (p *Parser) chunk() ast.Chunk {
        switch p.curToken.Type {
        case token.TEXT:
                return p.text()
        case token.EMPTY:
                return nil
        case token.LBRACES:
                return p.embedded(token.RBRACES)
        case token.LBRACESRAW:
                return p.embedded(token.RBRACESRAW)
        case token.IF:
                return p.ifDir()
        case token.FOR:
                return p.forDir()
        case token.EACH:
                return p.eachDir()
        case token.USE:
                return p.useDir()
        case token.RESERVE:
                return p.reserveDir()
        case token.INSERT:
                return p.insertDir()
        case token.BREAKIF:
                return p.breakifDir()
        case token.CONTINUEIF:
                return p.continueifDir()
        case token.COMPONENT:
                return p.compDir()
        case token.SLOT:
                return p.slotDir()
        case token.PASS, token.PASSIF:
                return p.passDir()
        case token.DUMP:
                return p.dumpDir()
        case token.BREAK:
                if illegal := p.illegalIfOutsideLoop(); illegal != nil {
                        return illegal
                }
                return ast.NewBreakDir(p.curToken)
        case token.CONTINUE:
                if illegal := p.illegalIfOutsideLoop(); illegal != nil {
                        return illegal
                }
                return ast.NewContinueDir(p.curToken)
        }

        return p.illegal()
}

func (p *Parser) embedded(tokenEnd token.TokenType) ast.Chunk {
        p.tokenEnd = tokenEnd
        embedded := ast.NewEmbedded(p.curToken, tokenEnd == token.RBRACESRAW)

        if p.peekTokenIs(tokenEnd) {
                pos := p.curToken.Pos
                pos.EndCol = p.peekToken.Pos.EndCol
                p.newError(pos, fail.ErrEmptyBraces)
                return nil
        }

        p.nextToken() // skip "{{"

        // Loop until we find the closing braces or reach the end of file
        for !p.curTokenIs(tokenEnd, token.EOF) {
                if segment := p.segment(); segment != nil {
                        embedded.Segments = append(embedded.Segments, segment)
                        p.finishSegment(tokenEnd)
                }
        }

        if p.curTokenIs(token.EOF) {
                p.newError(p.curToken.Pos, fail.ErrMissingClosingBraces, token.String(tokenEnd))
                return embedded
        }

        if p.peekTokenIs(tokenEnd) {
                p.nextToken() // skip "}}" or "!!}"
        }

        return embedded
}

// finishSegment consumes the ";" after a segment. If the segment is followed
// by anything else, it reports the error and skips to the next separator to
// avoid cascading errors.
func (p *Parser) finishSegment(tokenEnd token.TokenType) {
        if !p.curTokenIs(token.SEMI, tokenEnd, token.EOF) {
                p.reportUnexpectedTokenAfterSegment()
                p.skipUntil(token.SEMI, tokenEnd, token.EOF)
        }

        if p.curTokenIs(token.SEMI) {
                p.nextToken() // skip ";"
        }
}

// reportUnexpectedTokenAfterSegment reports why the current token cannot
// follow a segment.
func (p *Parser) reportUnexpectedTokenAfterSegment() {
        if p.prefixParseFns[p.curToken.Type] == nil {
                // Tokens that cannot start an expression, like an unknown operator,
                // get a dedicated error.
                p.illegalWith(fail.ErrIllegalToken, illegalLitText(p.curToken.Lit))
                return
        }

        p.newError(
                p.curToken.Pos,
                fail.ErrWrongPeekToken,
                token.String(token.SEMI),
                p.curToken.Lit,
        )
}

// skipUntil advances until the current token is one of the given types.
func (p *Parser) skipUntil(types ...token.TokenType) {
        for !p.curTokenIs(types...) {
                p.nextToken()
        }
}

func (p *Parser) peekIsStatement() bool {
        return p.peekTokenIs(token.ASSIGN, token.INC, token.DEC)
}

// segment parses individual segment
func (p *Parser) segment() ast.Segment {
        left := p.expression(LOWEST)
        if left == nil {
                p.nextToken()
                return nil
        }

        if !p.peekIsStatement() {
                p.nextToken()
                return left.(ast.Segment)
        }

        p.nextToken()
        stmt := p.statement(left)
        p.nextToken()

        if stmt == nil {
                return nil
        }

        return stmt.(ast.Segment)
}

func (p *Parser) registerPrefix(tokenType token.TokenType, fn prefixParseFn) {
        p.prefixParseFns[tokenType] = fn
}

func (p *Parser) registerInfix(tokenType token.TokenType, fn infixParseFn) {
        p.infixParseFns[tokenType] = fn
}

func (p *Parser) expectNonEmptyNameOn(node ast.Node) bool {
        if p.curToken.Lit == "" {
                p.newError(p.curToken.Pos, fail.ErrNameCannotBeEmpty, node.Tok().Lit)
                return false
        }
        return true
}

func (p *Parser) expectType(expectType token.TokenType) bool {
        if p.curToken.Type == expectType {
                return true
        }

        p.newError(
                p.curToken.Pos,
                fail.ErrWrongTokenType,
                token.String(expectType),
                token.String(p.curToken.Type),
        )

        return false
}

func (p *Parser) curTokenIs(tokens ...token.TokenType) bool {
        if len(tokens) == 1 {
                return tokens[0] == p.curToken.Type
        }
        return slices.Contains(tokens, p.curToken.Type)
}

func (p *Parser) peekTokenIs(tokens ...token.TokenType) bool {
        if len(tokens) == 1 {
                return tokens[0] == p.peekToken.Type
        }
        return slices.Contains(tokens, p.peekToken.Type)
}

func (p *Parser) peekPrecedence() int {
        result, ok := precedences[p.peekToken.Type]

        if !ok {
                return LOWEST
        }

        return result
}

func (p *Parser) expectPeek(tok token.TokenType) bool {
        if p.peekTokenIs(tok) {
                p.nextToken()
                return true
        }

        got := p.peekToken.Lit
        if len(got) > 100 {
                got = got[:100] + "..."
        }

        p.newError(p.peekToken.Pos, fail.ErrWrongPeekToken, token.String(tok), got)

        return false
}

func (p *Parser) newError(pos *position.Pos, msg string, args ...any) {
        newErr := fail.New(pos, p.file.Abs, fail.OriginPars, msg, args...)
        p.errors = append(p.errors, newErr)
}

func (p *Parser) nextToken() {
        p.curToken = p.peekToken
        p.peekToken = p.l.Next()
}

func (p *Parser) ident() ast.Expression {
        ident := ast.NewIdentExpr(p.curToken, p.curToken.Lit)
        if p.peekTokenIs(token.LPAREN) {
                return p.globalCallExpr(ident)
        }

        return ident
}

func (p *Parser) intExpr() ast.Expression {
        val, err := strconv.ParseInt(p.curToken.Lit, 10, 64)
        if err != nil {
                return p.illegalWith(fail.ErrCouldNotParseAs, p.curToken.Lit, "INT")
        }

        return ast.NewIntExpr(p.curToken, val)
}

func (p *Parser) floatExpr() ast.Expression {
        val, err := strconv.ParseFloat(p.curToken.Lit, 64)
        if err != nil {
                return p.illegalWith(fail.ErrCouldNotParseAs, p.curToken.Lit, "FLOAT")
        }

        return ast.NewFloatExpr(p.curToken, val)
}

func (p *Parser) nilExpr() ast.Expression {
        return ast.NewNilExpr(p.curToken)
}

func (p *Parser) strExpr() ast.Expression {
        return ast.NewStrExpr(p.curToken, p.curToken.Lit)
}

func (p *Parser) boolExpr() ast.Expression {
        return ast.NewBoolExpr(p.curToken, p.curTokenIs(token.TRUE))
}

func (p *Parser) arrExpr() ast.Expression {
        arr := ast.NewArrExpr(p.curToken)
        arr.Elements = p.expressionList(token.RBRACKET)
        arr.SetEndPosition(p.curToken.Pos)

        return arr
}

func (p *Parser) objExpr() ast.Expression {
        obj := ast.NewObjExpr(p.curToken)

        obj.Pairs = map[string]ast.Expression{}

        p.nextToken() // skip "{"

        if p.curTokenIs(token.RBRACE) {
                obj.SetEndPosition(p.curToken.Pos)
                return obj
        }

        for !p.curTokenIs(token.RBRACE, token.EOF) {
                key, ok := p.objKey()
                if !ok {
                        p.skipUntil(token.RBRACE, token.EOF)
                        break
                }

                obj.Pairs[key] = p.expression(LOWEST)

                if p.peekTokenIs(token.RBRACE) {
                        p.nextToken() // skip "}"
                        break
                }

                if p.peekTokenIs(token.COMMA) {
                        p.nextToken() // move to ","
                        p.nextToken() // skip ","
                        continue
                }

                p.newError(
                        p.peekToken.Pos,
                        fail.ErrWrongPeekToken,
                        token.String(token.RBRACE),
                        p.peekToken.Lit,
                )

                break
        }

        obj.SetEndPosition(p.curToken.Pos)

        return obj
}

// objKey reads an object key and the ":" that follows it.
// The second return reports whether the key was valid.
//
// Identifiers can also use the shorthand notation `{ key }`,
// which is the same as `{ key: key }`.
func (p *Parser) objKey() (string, bool) {
        if !p.curTokenIs(token.IDENT, token.STR) {
                p.newError(p.curToken.Pos, fail.ErrInvalidObjKey, p.curToken.Lit)
                return "", false
        }

        key := p.curToken.Lit

        switch {
        case p.peekTokenIs(token.COLON):
                p.nextToken() // move to ":"
                p.nextToken() // skip ":"
                return key, true
        case p.isShorthandKey():
                return key, true
        }

        p.newError(
                p.peekToken.Pos,
                fail.ErrWrongPeekToken,
                token.String(token.COLON),
                p.peekToken.Lit,
        )

        return "", false
}

// isShorthandKey reports whether the current identifier is a shorthand
// property, as in `{ key }` or `{ key, other: 1 }`.
func (p *Parser) isShorthandKey() bool {
        return p.curTokenIs(token.IDENT) && p.peekTokenIs(token.COMMA, token.RBRACE)
}

func (p *Parser) text() ast.Chunk {
        return ast.NewText(p.curToken)
}

func (p *Parser) assignStmt(left ast.Expression) ast.Statement {
        stmt := ast.NewAssignStmt(*left.Tok(), left)

        p.nextToken() // skip "="

        if p.curTokenIs(p.tokenEnd) {
                p.newError(p.curToken.Pos, fail.ErrExpectExprAfter, token.String(token.ASSIGN))
                return nil
        }

        stmt.Right = p.expression(LOWEST)
        stmt.SetEndPosition(p.curToken.Pos)

        return stmt
}

func (p *Parser) useDir() ast.Chunk {
        useDir := ast.NewUseDir(p.curToken)

        if !p.expectPeek(token.LPAREN) { // move to "("
                return p.illegal()
        }

        p.nextToken() // skip "("

        if !p.expectType(token.STR) {
                return p.illegal()
        }

        if !p.expectNonEmptyNameOn(useDir) {
                return p.illegal()
        }

        useDir.Name = ast.NewStrExpr(
                p.curToken,
                file.ReplacePathAlias(p.curToken.Lit, file.PathAliasUse),
        )

        if p.prog.UseDir != nil {
                p.newError(useDir.Pos(), fail.ErrOnlyOneUseDir)
        }

        if !p.expectPeek(token.RPAREN) { // move to ")"
                return p.illegal()
        }

        useDir.SetEndPosition(p.curToken.Pos)

        p.prog.UseDir = useDir

        return useDir
}

func (p *Parser) breakifDir() ast.Chunk {
        if illegal := p.illegalIfOutsideLoop(); illegal != nil {
                return illegal
        }

        dir := ast.NewBreakIfDir(p.curToken)

        if !p.expectPeek(token.LPAREN) { // move to "("
                return p.illegal()
        }

        p.nextToken() // skip "("

        dir.Cond = p.expression(LOWEST)

        if !p.expectPeek(token.RPAREN) { // move to ")"
                return p.illegal()
        }

        dir.SetEndPosition(p.curToken.Pos)

        return dir
}

func (p *Parser) continueifDir() ast.Chunk {
        if illegal := p.illegalIfOutsideLoop(); illegal != nil {
                return illegal
        }

        dir := ast.NewContinueIfDir(p.curToken)

        if !p.expectPeek(token.LPAREN) { // move to "("
                return p.illegal()
        }

        p.nextToken() // skip "("

        dir.Cond = p.expression(LOWEST)

        if !p.expectPeek(token.RPAREN) { // move to ")"
                return p.illegal()
        }

        dir.SetEndPosition(p.curToken.Pos)

        return dir
}

func (p *Parser) compDir() ast.Chunk {
        compDir := ast.NewCompDir(p.curToken)

        if illegal := p.compDirHeader(compDir); illegal != nil { // moves to ")"
                return illegal
        }

        p.nextToken() // skip ")"

        if p.curTokenIs(token.END) {
                return p.endCompDir(compDir)
        }

        if block := p.componentBlock(); block != nil {
                compDir.Passes = block.ExtractPassDirs()
                compDir.DefaultPass = block.ToDefaultPassDir(compDir.Name.Val)
        }

        if illegal := p.expectEnd("@component", compDir.Token.Pos.StartLine+1); illegal != nil {
                return illegal
        }

        return p.endCompDir(compDir)
}

// componentBlock parses a component body as a block with its own loop context.
//
// A component body is rendered inside the component's slot, not where it is
// written, so a loop surrounding the component does not surround the body at
// runtime. Resetting loopDepth makes loop control directives (@break, @breakif,
// @continue, @continueif) invalid here unless the body has its own loop.
//
//        @each(name in names)
//            @component('~user', { user })
//                @breakif(name == 'Sam') // error: not inside a loop
//            @end
//        @end
func (p *Parser) componentBlock() *ast.Block {
        outerLoopDepth := p.loopDepth
        p.loopDepth = 0
        defer func() { p.loopDepth = outerLoopDepth }()

        return p.block()
}

func (p *Parser) endCompDir(compDir *ast.CompDir) ast.Chunk {
        dup, times := ast.FindDuplicatePasses(compDir)
        if dup != nil && times > 0 {
                p.newError(dup.Pos(), fail.ErrDuplicatePass, dup.Name.Val, times, compDir.Name.Val)
                return p.illegal()
        }

        p.prog.Components = append(p.prog.Components, compDir)
        compDir.SetEndPosition(p.curToken.Pos)

        return compDir
}

func (p *Parser) compDirHeader(compDir *ast.CompDir) *ast.Illegal {
        if !p.expectPeek(token.LPAREN) { // move to "("
                return p.illegalUntil(token.RPAREN)
        }

        p.nextToken() // skip "("

        if !p.expectType(token.STR) {
                return p.illegal()
        }

        if !p.expectNonEmptyNameOn(compDir) {
                return ast.NewIllegalNode(p.curToken)
        }

        compDir.Name = ast.NewStrExpr(
                p.curToken,
                file.ReplacePathAlias(p.curToken.Lit, file.PathAliasComp),
        )

        if p.peekTokenIs(token.COMMA) {
                p.nextToken() // move to ","
                p.nextToken() // skip ","

                obj, ok := p.expression(LOWEST).(*ast.ObjExpr)
                if !ok {
                        p.newError(p.curToken.Pos, fail.ErrExpectedObjLit, p.curToken.Lit)
                        return nil
                }

                compDir.Argument = obj
        }

        if !p.expectPeek(token.RPAREN) { // move to ")"
                return p.illegalUntil(token.TEXT)
        }

        return nil
}

// slotDir parses an @slot inside a component file.
func (p *Parser) slotDir() ast.Chunk {
        slotDir := ast.NewSlotDir(p.curToken, p.file.Name)

        // Handle default @slot without name
        if !p.peekTokenIs(token.LPAREN) {
                slotDir.Name = ast.NewStrExpr(p.curToken, "")
                return p.endSlotDir(slotDir)
        }

        p.nextToken() // skip "@slot"
        p.nextToken() // skip "("

        // Handle named @slot with name
        slotDir.Name = ast.NewStrExpr(p.curToken, p.curToken.Lit)

        if !p.expectPeek(token.RPAREN) { // move to ")"
                return p.illegalUntil(token.END)
        }

        return p.endSlotDir(slotDir)
}

func (p *Parser) endSlotDir(slotDir *ast.SlotDir) ast.Chunk {
        name := slotDir.Name.Val
        if _, ok := p.prog.Slots[name]; ok {
                p.newError(slotDir.Name.Pos(), fail.ErrDuplicateSlots, name, p.file.Abs)
                return nil
        }

        slotDir.SetEndPosition(p.curToken.Pos)
        p.prog.Slots[name] = slotDir

        return slotDir
}

func (p *Parser) dumpDir() ast.Chunk {
        tok := p.curToken // "@dump"

        var args []ast.Expression

        if !p.expectPeek(token.LPAREN) { // move to "("
                return ast.NewIllegalNode(tok)
        }

        args = p.expressionList(token.RPAREN)

        dir := ast.NewDumpDir(tok, args)
        dir.SetEndPosition(p.curToken.Pos)

        return dir
}

func (p *Parser) passDir() ast.Chunk {
        passDir := ast.NewPassDir(p.curToken, nil)
        hasCondition := p.curToken.Type == token.PASSIF

        p.nextToken() // move to "("
        p.nextToken() // skip "("

        if hasCondition {
                passDir.Cond = p.expression(LOWEST)
        }

        if p.peekTokenIs(token.COMMA) {
                p.nextToken() // move to ","
        }

        if p.peekTokenIs(token.STR) {
                p.nextToken() // move to string
        }

        if !p.expectNonEmptyNameOn(passDir) {
                return ast.NewIllegalNode(p.curToken)
        }

        passDir.Name = ast.NewStrExpr(p.curToken, p.curToken.Lit)

        if !p.expectPeek(token.RPAREN) { // move to ")"
                return p.illegal()
        }

        p.nextToken() // skip ")"

        if !p.curTokenIs(token.END) {
                passDir.Block = p.block()
        }

        passDir.SetEndPosition(p.curToken.Pos)

        return passDir
}

func (p *Parser) reserveDir() ast.Chunk {
        reserveDir := ast.NewReserveDir(p.curToken)

        if !p.expectPeek(token.LPAREN) { // move to "("
                return p.illegal()
        }

        p.nextToken() // skip "("

        if !p.expectType(token.STR) {
                return p.illegal()
        }

        if !p.expectNonEmptyNameOn(reserveDir) {
                return p.illegal()
        }

        reserveDir.Name = ast.NewStrExpr(p.curToken, p.curToken.Lit)
        reserveDir.Fallback = p.reserveDirFallback()

        if !p.expectPeek(token.RPAREN) { // move to ")"
                return p.illegal()
        }

        reserveDir.SetEndPosition(p.curToken.Pos)

        // Check for duplicate reserve statements
        if _, ok := p.prog.Reserves[reserveDir.Name.Val]; ok {
                p.newError(
                        reserveDir.Name.Pos(),
                        fail.ErrDuplicateReserves,
                        reserveDir.Name.Val,
                        p.file.Abs,
                )
                return nil
        }

        p.prog.Reserves[reserveDir.Name.Val] = reserveDir

        return reserveDir
}

// reserveDirFallback parses the second argument (fallback value) after comma
// for @reserve statement. If there is no expression, set Fallback to Absent.
func (p *Parser) reserveDirFallback() ast.Expression {
        if !p.peekTokenIs(token.COMMA) {
                return ast.NewAbsent(p.curToken.Pos)
        }

        p.nextToken() // move to "," from string
        p.nextToken() // move to expression from ","

        return p.expression(LOWEST)
}

func (p *Parser) insertDir() ast.Chunk {
        insertDir := ast.NewInsertDir(p.curToken, p.file.Abs)

        illegal, done := p.insertDirHeader(insertDir) // moves to ")"
        if illegal != nil {
                return illegal
        }

        if done {
                return insertDir
        }

        p.nextToken() // skip ")"

        // Handle empty block
        if p.curTokenIs(token.END) {
                insertDir.SetEndPosition(p.curToken.Pos)
                return insertDir
        }

        insertDir.Block = p.block()

        // skip block and move to @end
        if !p.curTokenIs(token.END) {
                p.newError(insertDir.Pos(), fail.ErrInsertMustHaveContent, insertDir.Name.Val)
                return nil
        }

        insertDir.SetEndPosition(p.curToken.Pos)

        p.prog.Inserts[insertDir.Name.Val] = insertDir

        return insertDir
}

func (p *Parser) insertDirHeader(insertDir *ast.InsertDir) (*ast.Illegal, bool) {
        if !p.expectPeek(token.LPAREN) { // move to "("
                return p.illegal(), false
        }

        p.nextToken() // skip "("

        if !p.expectType(token.STR) {
                return p.illegal(), false
        }

        if !p.expectNonEmptyNameOn(insertDir) {
                return p.illegal(), false
        }

        insertDir.Name = ast.NewStrExpr(p.curToken, p.curToken.Lit)

        if ok := p.hasDuplicateInserts(insertDir); ok {
                return p.illegal(), false
        }

        illegal, done := p.insertDirArgument(insertDir)

        if illegal != nil {
                return illegal, false
        }

        if done {
                return nil, true
        }

        if !p.expectPeek(token.RPAREN) { // move to ")"
                return p.illegal(), false
        }

        return nil, false
}

func (p *Parser) insertDirArgument(insertDir *ast.InsertDir) (*ast.Illegal, bool) {
        if !p.peekTokenIs(token.COMMA) {
                insertDir.Argument = ast.NewAbsent(p.curToken.Pos)
                return nil, false
        }

        p.nextToken() // skip insert name
        p.nextToken() // skip ","
        insertDir.Argument = p.expression(LOWEST)

        if !p.expectPeek(token.RPAREN) { // move to ")"
                return p.illegal(), false
        }

        insertDir.SetEndPosition(p.curToken.Pos)

        p.prog.Inserts[insertDir.Name.Val] = insertDir

        return nil, true
}

func (p *Parser) hasDuplicateInserts(insertDir *ast.InsertDir) bool {
        if _, ok := p.prog.Inserts[insertDir.Name.Val]; !ok {
                return false
        }
        p.newError(insertDir.Name.Pos(), fail.ErrDuplicateInserts, insertDir.Name.Val)
        return true
}

func (p *Parser) indexExpr(left ast.Expression) ast.Expression {
        expr := ast.NewIndexExpr(*left.Tok(), left)

        p.nextToken() // skip "["

        expr.Index = p.expression(LOWEST)

        if !p.expectPeek(token.RBRACKET) { // move to "]"
                return p.illegal()
        }

        expr.SetEndPosition(p.curToken.Pos)

        return expr
}

func (p *Parser) incStmt(left ast.Expression) ast.Statement {
        return ast.NewIncStmt(p.curToken, left)
}

func (p *Parser) decStmt(left ast.Expression) ast.Statement {
        return ast.NewDecStmt(p.curToken, left)
}

func (p *Parser) dotExpr(left ast.Expression) ast.Expression {
        expr := ast.NewDotExpr(*left.Tok(), left)

        if !p.expectPeek(token.IDENT) { // skip "." and move to identifier
                return p.illegal()
        }

        if p.peekTokenIs(token.LPAREN) {
                return p.callExpr(left)
        }

        expr.Key = ast.NewIdentExpr(p.curToken, p.curToken.Lit)

        return expr
}

func (p *Parser) globalCallExpr(ident *ast.IdentExpr) ast.Expression {
        name := ast.GlobalFuncName(ident.Name)

        rules, exists := ast.GlobalFunctions[name]
        if !exists {
                p.newError(p.peekToken.Pos, fail.ErrGlobalFuncMissing, ident.Name)
                return ast.NewIllegalNode(p.curToken)
        }

        expr := ast.NewGlobalCallExpr(p.curToken, name)

        if !p.expectPeek(token.LPAREN) { // move to "("
                return p.illegal()
        }

        expr.Arguments = p.expressionList(token.RPAREN)
        expr.SetEndPosition(p.curToken.Pos)
        argsLen := len(expr.Arguments)

        if argsLen < rules.Min {
                p.newError(expr.Pos(), fail.ErrGlobalFuncFewArgs, name, rules.Min, argsLen)
                return ast.NewIllegalNode(p.curToken)
        }

        if argsLen > rules.Max {
                p.newError(expr.Pos(), fail.ErrGlobalFuncLotsOfArgs, name, rules.Max, argsLen)
                return ast.NewIllegalNode(p.curToken)
        }

        return expr
}

func (p *Parser) callExpr(receiver ast.Expression) ast.Expression {
        ident := ast.NewIdentExpr(p.curToken, p.curToken.Lit)
        expr := ast.NewCallExpr(p.curToken, receiver, ident)

        if !p.expectPeek(token.LPAREN) { // move to "("
                return p.illegal()
        }

        expr.Arguments = p.expressionList(token.RPAREN)
        expr.SetEndPosition(p.curToken.Pos)

        return expr
}

func (p *Parser) infixExpr(left ast.Expression) ast.Expression {
        opTok := p.curToken
        expr := ast.NewInfixExpr(*left.Tok(), left, opTok.Lit)

        precedence := precedences[opTok.Type]

        if p.peekTokenIs(p.tokenEnd) {
                p.newError(opTok.Pos, fail.ErrExpectExprAfter, opTok.Lit)
                return ast.NewIllegalNode(opTok)
        }

        p.nextToken() // skip operator

        expr.Right = p.expression(precedence)
        expr.SetEndPosition(p.curToken.Pos)

        return expr
}

func (p *Parser) ternaryExpr(left ast.Expression) ast.Expression {
        expr := ast.NewTernaryExpr(*left.Tok(), left)

        p.nextToken() // skip "?"

        expr.IfExpr = p.expression(LOWEST)

        if !p.expectPeek(token.COLON) { // move to ":"
                return p.illegal()
        }

        p.nextToken() // skip ":"

        expr.ElseExpr = p.expression(LOWEST)
        expr.SetEndPosition(p.curToken.Pos)

        return expr
}

func (p *Parser) ifDir() ast.Chunk {
        dir := ast.NewIfDir(p.curToken)

        if illegal := p.ifDirHeader(dir); illegal != nil { // skips ")"
                return illegal
        }

        dir.IfBlock = p.block()

        // Only look for @elseif/@else when the block wasn't closed by @end.
        if !p.curTokenIs(token.END) {
                if illegal := p.ifDirBranches(dir); illegal != nil {
                        return illegal
                }
        }

        dir.SetEndPosition(p.curToken.Pos)

        return dir
}

// ifDirBranches parses the @elseif/@else part and verifies the closing @end.
// It returns a non-nil node only when parsing failed.
func (p *Parser) ifDirBranches(dir *ast.IfDir) ast.Chunk {
        for p.curTokenIs(token.ELSEIF) {
                elseifDir, illegal := p.elseifDir()
                if illegal != nil {
                        return illegal
                }

                dir.ElseifDirs = append(dir.ElseifDirs, elseifDir)
        }

        dir.ElseBlock = p.elseBlock()

        return p.expectEnd("@if", dir.Token.Pos.StartLine+1)
}

// expectEnd checks that the current token is @end. It returns nil when the
// directive is properly closed and an illegal node otherwise. When the file
// ends before @end, it reports the missing @end for the directive.
func (p *Parser) expectEnd(dirName string, startLine uint) ast.Chunk {
        if p.curTokenIs(token.END) {
                return nil
        }

        if p.curTokenIs(token.EOF) {
                p.newError(p.curToken.Pos, fail.ErrMissingEnd, dirName, startLine)
                return ast.NewIllegalNode(p.curToken)
        }

        return p.illegal()
}

func (p *Parser) ifDirHeader(ifDir *ast.IfDir) *ast.Illegal {
        if !p.expectPeek(token.LPAREN) { // move to "("
                return p.illegal()
        }

        p.nextToken() // skip "("

        ifDir.Cond = p.expression(LOWEST)

        if !p.expectPeek(token.RPAREN) { // move to ")"
                return p.illegal()
        }

        p.nextToken() // skip ")"

        return nil
}

func (p *Parser) elseifDir() (*ast.ElseIfDir, *ast.Illegal) {
        dir := ast.NewElseIfDir(p.curToken)

        p.nextToken() // skip "@elseif"
        p.nextToken() // skip "("

        dir.Cond = p.expression(LOWEST)

        if !p.expectPeek(token.RPAREN) { // move to ")"
                return nil, p.illegal()
        }

        p.nextToken() // skip ")"

        dir.Block = p.block()
        dir.SetEndPosition(p.curToken.Pos)

        return dir, nil
}

func (p *Parser) elseBlock() *ast.Block {
        hasElse := p.curTokenIs(token.ELSE)
        if hasElse {
                p.nextToken() // skip "@else"
        }

        // "@else" immediately followed by "@end" is an empty else block.
        if hasElse && p.curTokenIs(token.END) {
                block := ast.NewBlock(p.curToken)
                block.SetEndPosition(p.curToken.Pos)
                return block
        }

        // No else block here: either there was no "@else", or it is followed
        // by another "@else".
        if p.curTokenIs(token.ELSE, token.END) {
                return nil
        }

        block := p.block()

        // "@elseif" is not allowed after "@else". The peek token takes
        // precedence when choosing the error position.
        switch {
        case p.peekTokenIs(token.ELSEIF):
                p.newError(p.peekToken.Pos, fail.ErrElseifCannotFollowElse)
                return nil
        case p.curTokenIs(token.ELSEIF):
                p.newError(p.curToken.Pos, fail.ErrElseifCannotFollowElse)
                return nil
        }

        if block == nil {
                return nil
        }

        block.SetEndPosition(p.curToken.Pos)

        return block
}

func (p *Parser) forDir() ast.Chunk {
        dir := ast.NewForDir(p.curToken)

        if illegal := p.forDirHeader(dir); illegal != nil { // skips ")"
                return illegal
        }

        p.loopDepth++
        defer func() { p.loopDepth-- }()

        dir.Block = p.block()

        if p.curTokenIs(token.END) {
                dir.SetEndPosition(p.curToken.Pos)
                return dir
        }

        dir.ElseBlock = p.elseBlock()

        if illegal := p.expectEnd("@for", dir.Token.Pos.StartLine+1); illegal != nil {
                return illegal
        }

        dir.SetEndPosition(p.curToken.Pos)

        return dir
}

func (p *Parser) forDirHeader(forDir *ast.ForDir) *ast.Illegal {
        if !p.expectPeek(token.LPAREN) { // move to "("
                return p.illegalUntil(token.END)
        }

        forDir.Init = p.parserForDirInit()
        if forDir.Init == nil {
                p.skipUntil(token.END, token.EOF)
                return nil
        }

        if !p.expectPeek(token.SEMI) { // move to ";"
                return p.illegalUntil(token.END)
        }

        forDir.Cond = p.parserForDirCond()

        if !p.expectPeek(token.SEMI) { // move to ";"
                return p.illegalUntil(token.END)
        }

        post := p.parseForDirPost()
        if post == nil {
                return nil
        }

        forDir.Post = post

        if !p.expectPeek(token.RPAREN) { // move to ")"
                return p.illegalUntil(token.END)
        }

        p.nextToken() // skip ")"

        return nil
}

func (p *Parser) parserForDirInit() ast.Statement {
        if p.peekTokenIs(token.SEMI) {
                return ast.NewAbsent(p.curToken.Pos)
        }

        p.nextToken() // move to first token of init statement
        left := p.expression(LOWEST)
        p.nextToken() // move to =/++/--

        stmt := p.statement(left)
        if stmt != nil {
                return stmt
        }

        p.newError(left.Pos(), fail.ErrForLoopExpectInitStmt, left.String())
        return nil
}

func (p *Parser) parserForDirCond() ast.Expression {
        if p.peekTokenIs(token.SEMI) {
                return ast.NewAbsent(p.curToken.Pos)
        }

        p.nextToken() // skip ";"

        return p.expression(LOWEST)
}

func (p *Parser) parseForDirPost() ast.Statement {
        if p.peekTokenIs(token.RPAREN) {
                return ast.NewAbsent(p.curToken.Pos)
        }

        p.nextToken() // skip ";"
        left := p.expression(LOWEST)
        p.nextToken() // move to ++/--

        if p.curTokenIs(token.RPAREN) {
                p.newError(left.Pos(), fail.ErrForLoopExpectStmt, left.String())
                p.skipUntil(token.END, token.EOF)
                return nil
        }

        stmt := p.statement(left)
        if stmt == nil {
                p.newError(p.curToken.Pos, fail.ErrForLoopExpectRParen)
                p.skipUntil(token.END, token.EOF)
                return nil
        }

        return stmt
}

func (p *Parser) eachDir() ast.Chunk {
        dir := ast.NewEachDir(p.curToken)

        if illegal := p.eachDirHeader(dir); illegal != nil { // skips ")"
                return illegal
        }

        p.loopDepth++
        defer func() { p.loopDepth-- }()

        dir.Block = p.block()

        if p.curTokenIs(token.END) {
                dir.SetEndPosition(p.curToken.Pos)
                return dir
        }

        dir.ElseBlock = p.elseBlock()

        if illegal := p.expectEnd("@each", dir.Token.Pos.StartLine+1); illegal != nil {
                return illegal
        }

        dir.SetEndPosition(p.curToken.Pos)

        return dir
}

func (p *Parser) eachDirHeader(dir *ast.EachDir) *ast.Illegal {
        if !p.expectPeek(token.LPAREN) { // move to "("
                return p.illegal()
        }

        p.nextToken() // skip "("

        if !p.expectType(token.IDENT) {
                return p.illegal()
        }

        dir.Var = ast.NewIdentExpr(p.curToken, p.curToken.Lit)

        if !p.expectPeek(token.IN) { // move to "in"
                return p.illegal()
        }

        p.nextToken() // skip "in"

        dir.Arr = p.expression(LOWEST)

        if !p.expectPeek(token.RPAREN) { // move to ")"
                return p.illegal()
        }

        p.nextToken() // skip ")"

        return nil
}

func (p *Parser) block() *ast.Block {
        if p.curTokenIs(token.ELSE, token.ELSEIF, token.END) {
                return nil
        }

        block := ast.NewBlock(p.curToken)
        block.SetEndPosition(p.curToken.Pos)

        for !p.curTokenIs(token.END) && !p.curTokenIs(token.EOF) {
                chunk := p.chunk()
                block.SetEndPosition(p.curToken.Pos)

                if chunk != nil {
                        block.Chunks = append(block.Chunks, chunk)
                }

                if p.peekTokenIs(token.ELSE, token.ELSEIF, token.END) {
                        p.nextToken() // skip chunk
                        break
                }

                p.nextToken() // skip chunk
        }

        return block
}

func (p *Parser) statement(left ast.Expression) ast.Statement {
        switch p.curToken.Type {
        case token.ASSIGN:
                return p.assignStmt(left)
        case token.INC:
                return p.incStmt(left)
        case token.DEC:
                return p.decStmt(left)
        }

        return nil
}

func (p *Parser) expression(precedence int) ast.Expression {
        prefix := p.prefixParseFns[p.curToken.Type]

        if prefix == nil {
                return p.illegal()
        }

        leftExpr := prefix()

        for !p.peekTokenIs(p.tokenEnd, token.SEMI, token.RPAREN) && precedence < p.peekPrecedence() {
                infix := p.infixParseFns[p.peekToken.Type]

                if infix == nil {
                        return leftExpr
                }

                p.nextToken()

                leftExpr = infix(leftExpr)
        }

        if leftExpr != nil {
                leftExpr.SetEndPosition(p.curToken.Pos)
        }

        return leftExpr
}

func (p *Parser) prefixExpr() ast.Expression {
        prefixTok := p.curToken

        p.nextToken() // skip prefix operator

        if expr, ok := p.negativeIntLiteral(prefixTok); ok {
                return expr
        }

        expr := ast.NewPrefixExpr(prefixTok, prefixTok.Lit)
        expr.Right = p.expression(PREFIX)
        expr.SetEndPosition(p.curToken.Pos)

        return expr
}

// negativeIntLiteral handles "-9223372036854775808", the one literal whose
// magnitude doesn't fit in int64 but whose negative value does. Everything
// else goes through the regular prefix-expression path.
func (p *Parser) negativeIntLiteral(opTok token.Token) (ast.Expression, bool) {
        if opTok.Type != token.SUB || !p.curTokenIs(token.INT) {
                return nil, false
        }

        lit := p.curToken.Lit

        // Happy path when the literal is not the special case.
        val, err := strconv.ParseInt("-"+lit, 10, 64)
        if err != nil || val != math.MinInt64 {
                return nil, false
        }

        // The literal's position spans from the "-" to the end of the number.
        pos := *p.curToken.Pos
        pos.StartLine = opTok.Pos.StartLine
        pos.StartCol = opTok.Pos.StartCol

        tok := token.Token{Type: token.INT, Lit: lit, Pos: &pos}

        return ast.NewIntExpr(tok, val), true
}

func (p *Parser) groupedExpr() ast.Expression {
        exprTok := p.curToken

        p.nextToken() // skip "("

        expr := p.expression(LOWEST)
        if expr == nil {
                return p.illegal()
        }

        expr.SetTok(exprTok)

        if !p.expectPeek(token.RPAREN) { // move to ")"
                return p.illegal()
        }

        return expr
}

func (p *Parser) expressionList(endTok token.TokenType) []ast.Expression {
        firstTok := p.curToken
        var exprs []ast.Expression

        if p.peekTokenIs(endTok) {
                p.nextToken() // skip endTok token
                return exprs
        }

        if p.peekTokenIs(token.END) {
                p.newError(p.peekToken.Pos, fail.ErrExpectExprAfter, firstTok.Lit)
                return nil
        }

        p.nextToken() // move to first expression

        exprs = append(exprs, p.expression(LOWEST))

        for p.peekTokenIs(token.COMMA) && !p.curTokenIs(token.EOF) {
                p.nextToken() // move to ","

                // break when has a trailing comma
                if p.peekTokenIs(endTok) {
                        break
                }

                p.nextToken() // skip ","
                exprs = append(exprs, p.expression(LOWEST))
        }

        if !p.expectPeek(endTok) { // move to endTok
                exprs = append(exprs, ast.NewIllegalNode(p.curToken))
                return exprs
        }

        return exprs
}

func (p *Parser) illegal() *ast.Illegal {
        if p.curTokenIs(token.UNCLOSED_COMMENT) {
                return p.illegalWith(fail.ErrMissingClosingBraces, "--}}")
        }
        return p.illegalWith(fail.ErrIllegalToken, illegalLitText(p.curToken.Lit))
}

// illegalLitText returns a single illegal character as it is. Invisible
// characters and invalid bytes are shown by their code point instead, so
// they don't look like a regular space or a replacement character.
func illegalLitText(lit string) string {
        r, size := utf8.DecodeRuneInString(lit)
        if size != len(lit) {
                return lit
        }

        if r == utf8.RuneError && size == 1 {
                return fmt.Sprintf("0x%02X", lit[0])
        }

        if unicode.IsPrint(r) {
                return lit
        }

        return fmt.Sprintf("U+%04X", r)
}

// illegalWith records an error at the current token and returns an illegal
// node for it.
func (p *Parser) illegalWith(msg string, args ...any) *ast.Illegal {
        p.newError(p.curToken.Pos, msg, args...)
        return ast.NewIllegalNode(p.curToken)
}

// illegalIfOutsideLoop returns an Illegal node (and reports an error) when a
// loop control directive is used outside of a loop. It returns nil when the
// directive is allowed.
func (p *Parser) illegalIfOutsideLoop() *ast.Illegal {
        if p.loopDepth > 0 {
                return nil
        }

        return p.illegalWith(fail.ErrLoopControlOutsideLoop, p.curToken.Lit)
}

func (p *Parser) illegalUntil(tok token.TokenType) *ast.Illegal {
        p.skipUntil(tok, token.EOF)
        return p.illegal()
}

type parseOpts struct {
        chunksCount int
        checkErrors bool
}

var defaultParseOpts = parseOpts{
        chunksCount: 1,
        checkErrors: true,
}

func parseChunks(inp string, opts parseOpts) ([]ast.Chunk, error) {
        l := lexer.New(inp)
        p := New(l, nil)
        prog := p.ParseProgram()

        if opts.checkErrors && p.HasErrors() {
                return nil, p.Errors()[0].Error()
        }

        if len(prog.Chunks) != opts.chunksCount {
                return nil, fmt.Errorf(
                        "program must have %d chunks but got %d for input %q",
                        opts.chunksCount,
                        len(prog.Chunks),
                        inp,
                )
        }

        return prog.Chunks, nil
}

func parseEmbedded[T ast.Segment](inp string) (T, error) {
        var zero T

        chunks, err := parseChunks(inp, defaultParseOpts)
        if err != nil {
                return zero, err
        }

        embedded, ok := chunks[0].(*ast.Embedded)
        if !ok {
                return zero, fmt.Errorf("chunks[0] is not an Embedded, got %T", chunks[0])
        }

        if len(embedded.Segments) != 1 {
                return zero, fmt.Errorf(
                        "embedded.Segments must contain 1 segment, got %d",
                        len(embedded.Segments),
                )
        }

        segment, ok := embedded.Segments[0].(T)
        if !ok {
                return zero, fmt.Errorf(
                        "embedded.Segments[0] is not %T, got %T",
                        zero,
                        embedded.Segments[0],
                )
        }

        return segment, nil
}

func parseEmbeddedSegments(inp string, opts parseOpts) ([]ast.Segment, error) {
        chunks, err := parseChunks(inp, opts)
        if err != nil {
                return nil, err
        }

        embedded, ok := chunks[0].(*ast.Embedded)
        if !ok {
                return nil, fmt.Errorf("chunks[0] is not Embedded, got %T", chunks[0])
        }

        return embedded.Segments, nil
}

func parseDirective[T ast.Chunk](inp string, opts parseOpts) (T, error) {
        var zero T

        chunks, err := parseChunks(inp, opts)
        if err != nil {
                return zero, err
        }

        dir, ok := chunks[0].(T)
        if !ok {
                return zero, fmt.Errorf("chunks[0] is not an %T, got %T", zero, chunks[0])
        }

        return dir, nil
}

// parseChunkInLoop parses a chunk that is only valid inside a loop. It wraps the
// input in an "@each" loop and returns the first chunk of the loop's block.
func parseChunkInLoop[T ast.Chunk](inp string, opts parseOpts) (T, error) {
        var zero T

        chunks, err := parseChunks("@each(x in [1])"+inp+"@end", opts)
        if err != nil {
                return zero, err
        }

        eachDir, ok := chunks[0].(*ast.EachDir)
        if !ok {
                return zero, fmt.Errorf("chunks[0] is not an EachDir, got %T", chunks[0])
        }

        if len(eachDir.Block.Chunks) == 0 {
                return zero, fmt.Errorf("loop block has no chunks for input %q", inp)
        }

        chunk, ok := eachDir.Block.Chunks[0].(T)
        if !ok {
                return zero, fmt.Errorf(
                        "loop block chunk is not %T, got %T",
                        zero,
                        eachDir.Block.Chunks[0],
                )
        }

        return chunk, nil
}

func testInfixExpr(expr ast.Expression, left any, op string, right any) error {
        infixExpr, ok := expr.(*ast.InfixExpr)
        if !ok {
                return fmt.Errorf("expr is not an InfixExpr, got %T", expr)
        }

        if err := testLiteralExpr(infixExpr.Left, left); err != nil {
                return err
        }

        if infixExpr.Op != op {
                return fmt.Errorf("infixExpr.Op is not %s, got %s", op, infixExpr.Op)
        }

        if err := testLiteralExpr(infixExpr.Right, right); err != nil {
                return err
        }

        return nil
}

func testTokPosition(actual, expect *position.Pos) error {
        if expect.StartLine != actual.StartLine {
                return fmt.Errorf("expect.StartLine is not %d, got %d", expect.StartLine, actual.StartLine)
        }

        if expect.EndLine != actual.EndLine {
                return fmt.Errorf("expect.EndLine is not %d, got %d", expect.EndLine, actual.EndLine)
        }

        if expect.StartCol != actual.StartCol {
                return fmt.Errorf("expect.StartCol is not %d, got %d", expect.StartCol, actual.StartCol)
        }

        if expect.EndCol != actual.EndCol {
                return fmt.Errorf("expect.EndCol is not %d, got %d", expect.EndCol, actual.EndCol)
        }

        return nil
}

func testIntExpr(expr ast.Expression, val int64) error {
        integer, ok := expr.(*ast.IntExpr)
        if !ok {
                return fmt.Errorf("expr is not an IntExpr, got %T", expr)
        }

        if integer.Val != val {
                return fmt.Errorf("integer.Val is not %d, got %d", val, integer.Val)
        }

        if integer.Tok().Lit != strconv.FormatInt(val, 10) {
                return fmt.Errorf("integer.Tok().Lit is not %d, got %s", val, integer.Tok().Lit)
        }

        return nil
}

func testFloatExpr(expr ast.Expression, val float64) error {
        float, ok := expr.(*ast.FloatExpr)
        if !ok {
                return fmt.Errorf("expr is not a FloatExpr, got %T", expr)
        }

        if float.Val != val {
                return fmt.Errorf("float.Val is not %f, got %f", val, float.Val)
        }

        if float.String() != utils.FloatToStr(val) {
                return fmt.Errorf("float.String() is not %f, got %s", val, float)
        }

type Pos struct {
        StartLine uint
        StartCol  uint
        EndLine   uint
        EndCol    uint
}

func (p Pos) Contains(line uint, col uint) bool {
        // Line is out of range
        if line < p.StartLine || line > p.EndLine {
                return false
        }

        // On start line, check if cursor is after start column
        if line == p.StartLine && col < p.StartCol {
                return false
        }

        // On end line, check if cursor is before end column
        if line == p.EndLine && col > p.EndCol {
                return false
        }

        return true
}

func (p Pos) Line() uint {
        return p.StartLine + 1
}

// Col returns the start column position display.
func (p Pos) Col() uint {
        return p.StartCol + 1
}

        }

func Start(in io.Reader, out io.Writer) error {
        scanner := bufio.NewScanner(in)
        scope := value.NewScope(nil)

        for {
                fmt.Print(PROMPT)

                scanned := scanner.Scan()
                if !scanned {
                        return nil
                }

                l := lexer.New(scanner.Text())
                p := parser.New(l, nil)
                prog := p.ParseProgram()

                if len(p.Errors()) != 0 {
                        if err := printParserErrors(out, p.Errors()); err != nil {
                                return err
                        }
                        continue
                }

                e := evaluator.New(nil, nil, nil)
                ctx := evaluator.NewContext(scope, prog.AbsPath)
                evaluated := e.Eval(prog, ctx)

                if _, err := io.WriteString(out, evaluated.String()+"\n"); err != nil {
                        return err
                }
        }
}

func printParserErrors(out io.Writer, errors []*fail.Error) error {
        if _, err := io.WriteString(out, "Textwire errors:\n"); err != nil {
                return err
        }

        for _, err := range errors {
                if _, err := io.WriteString(out, "\t"+err.String()+"\n"); err != nil {
                        return err
                }
        }

        return nil
}

// FloatToStr converts float64 to string using a
// precision of -1 to preserve the exact value
func FloatToStr(f float64) string {
        // Infinities and NaN have no decimal part, so don't add ".0" to them.
        if math.IsInf(f, 0) || math.IsNaN(f) {
                return strconv.FormatFloat(f, 'g', -1, 64)
        }

        // Use 'floatStr' format first to see if it uses scientific notation
        floatStr := strconv.FormatFloat(f, 'g', -1, 64)

        // If 'floatStr' format uses scientific notation and the number is very large, use it
        if (strings.Contains(floatStr, "e") || strings.Contains(floatStr, "E")) &&
                (f > 1e20 || f < -1e20) {
                return floatStr
        }

        // Otherwise use 'f' format for decimal notation
        str := strconv.FormatFloat(f, 'f', -1, 64)

        // If no decimal point, add .0
        if !strings.Contains(str, ".") {
                str += ".0"
        }

        return str
}

func ToCamel(s string) string {
        s = strings.TrimSpace(s)
        if s == "" {
                return s
        }

        var n strings.Builder
        n.Grow(len(s))

        capNext := false
        prevIsCap := false

        for i, v := range []byte(s) {
                vIsCap := v >= 'A' && v <= 'Z'
                vIsLow := v >= 'a' && v <= 'z'

                if capNext {
                        if vIsLow {
                                v += 'A'
                                v -= 'a'
                        }
                } else if i == 0 {
                        if vIsCap {
                                v += 'a'
                                v -= 'A'
                        }
                } else if prevIsCap && vIsCap {
                        v += 'a'
                        v -= 'A'
                }

                prevIsCap = vIsCap

                if vIsCap || vIsLow {
                        n.WriteByte(v)
                        capNext = false
                } else if vIsNum := v >= '0' && v <= '9'; vIsNum {
                        n.WriteByte(v)
                        capNext = true
                } else {
                        capNext = v == '_' || v == ' ' || v == '-' || v == '.'
                }
        }
        return n.String()
}

// the cycle check for child arrays and objects.

        return a.joinElements(htmlOf)

// joinElements converts each element with elemOf and joins them with commas.
func (a *Arr) joinElements(toString func(Literal) string) string {
        if len(a.Elements) == 0 {
                return ""
        }

        var out bytes.Buffer
        for _, elem := range a.Elements {
                out.WriteString(toString(elem))
                out.WriteString(", ")
        }

        if out.Len() > 1 {
                out.Truncate(out.Len() - 2)
        }

        return out.String()
}

func (a *Arr) Dump(indent int) string {
        if HasCycle(a) {
                return circularReference
        }
        return a.dumpNoCycleCheck(indent)
}

func (a *Arr) dumpNoCycleCheck(indent int) string {
        spaces := strings.Repeat("  ", indent)
        indent += 1

        var out bytes.Buffer

        fmt.Fprintf(&out, `<span style="%s">array:%d </span>`, DUMP_META, len(a.Elements))
        fmt.Fprintf(&out, `<span style="%s">[</span>`, DUMP_BRACE)

        if len(a.Elements) == 0 {
                spaces = ""
        } else {
                out.WriteByte('\n')
        }

        insideSpaces := strings.Repeat("  ", indent)

        for _, elem := range a.Elements {
                out.WriteString(insideSpaces)
                out.WriteString(dumpOf(elem, indent))
                out.WriteString(",\n")
        }

        res := out.String()

        // take last 8 characters of the string
        lastChar := res[len(res)-8:]

        if lastChar == "}</span>" {
                res += "\n"
        }

        return res + spaces + fmt.Sprintf(`<span style="%s">]</span>`, DUMP_BRACE)
}

func (a *Arr) JSON() (string, error) {
        if HasCycle(a) {
                return "", fmt.Errorf(fail.ErrCircularValue, a.Type())
        }
        return a.jsonNoCycleCheck()
}

func (a *Arr) jsonNoCycleCheck() (string, error) {
        var out strings.Builder
        out.Grow(len(a.Elements) + 2)

        out.WriteByte('[')

        for i := range a.Elements {
                if i > 0 {
                        out.WriteByte(',')
                }

                val, err := jsonOf(a.Elements[i])
                if err != nil {
                        return "", err
                }

                out.WriteString(val)
        }

        out.WriteByte(']')

        return out.String(), nil
}

        if HasCycle(a) {
                return nil
        }
        return a.nativeNoCycleCheck()
}

func (a *Arr) nativeNoCycleCheck() any {
        var vals []any
        for _, elem := range a.Elements {
                vals = append(vals, nativeOf(elem))
        }
        return vals
}

        return t == a.Type()
}

// circularReference is rendered by String and Dump when a value contains a
// cycle. JSON returns an error instead.
const circularReference = "[circular reference]"

const (
        unvisited visitState = iota
        visiting
        done
)

type visitState uint8

// References reports whether target is reachable from root through arrays and
// objects. It walks root's arrays and objects depth first and returns true
// if it reaches target. It has no depth limit.
func References(root, target Literal) bool {
        if root == nil || target == nil {
                return false
        }

        seen := map[Literal]bool{}
        stack := []Literal{root}

        for len(stack) > 0 {
                // The last item is the top of the stack
                current := stack[len(stack)-1]

                // Removes current item by shortening the slice by one
                stack = stack[:len(stack)-1]

                if current == target {
                        return true
                }

                kids, ok := childValues(current)
                if !ok || seen[current] {
                        continue
                }

                seen[current] = true
                stack = append(stack, kids...)
        }

        return false
}

// HasCycle reports whether root contains itself, directly or through nested
// arrays and objects. For example, arr[0] = arr makes arr contain itself.
func HasCycle(root Literal) bool {
        return cycleFrom(root, map[Literal]visitState{})
}

// cycleFrom walks depth first. An array or object is marked visiting while its
// elements are walked, so reaching it again means a cycle. Once marked done, it
// can't be part of a cycle, so it's skipped.
func cycleFrom(v Literal, state map[Literal]visitState) bool {
        kids, ok := childValues(v)
        if !ok {
                return false
        }

        switch state[v] {
        case visiting:
                return true
        case done:
                return false
        }

        state[v] = visiting

        for _, kid := range kids {
                if cycleFrom(kid, state) {
                        return true
                }
        }

        state[v] = done

        return false
}

// childValues returns the elements of an array or the values of an object. Object
// keys are plain strings, so they aren't included. The boolean is false when v
// isn't a non-nil array or object.
func childValues(v Literal) ([]Literal, bool) {
        switch v := v.(type) {
        case *Arr:
                if v == nil {
                        return nil, false
                }
                return v.Elements, true
        case *Obj:
                if v == nil {
                        return nil, false
                }
                return objectValues(v), true
        default:

                values = append(values, value)
        }
        return values
}

// emptyGlobal is shared by every scope without global data. It's frozen, so
// sharing it is safe.
var emptyGlobal = func() *Obj {
        obj := NewObj(nil)
        obj.frozen = true
        return obj
}()

// Freeze marks v and everything reachable from it as read-only.
func Freeze(v Literal) {
        freezeWalk(v, map[Literal]bool{})
}

// NewFrozenGlobal converts native global data into a frozen object.
func NewFrozenGlobal(data map[string]any) (*Obj, *fail.Error) {
        if data == nil {
                return emptyGlobal, nil
        }

        lit := NativeToValue(data)
        if errObj, ok := lit.(*Error); ok {
                return nil, errObj.Err
        }

        // NativeToValue returns an *Obj for any map, so this is only a guard.
        obj, ok := lit.(*Obj)
        if !ok {
                return nil, fail.New(nil, "", fail.OriginTpl, fail.ErrUnknownType, lit.Type())
        }

        Freeze(obj)

        return obj, nil
}

// freezeWalk freezes arrays and objects depth first. The seen map keeps cycles
// from recursing forever.
func freezeWalk(v Literal, seen map[Literal]bool) {
        kids, ok := childValues(v)
        if !ok || seen[v] {
                return
        }

        seen[v] = true

        switch val := v.(type) {
        case *Obj:
                val.frozen = true
        case *Arr:
                val.frozen = true
        }

        for _, kid := range kids {
                freezeWalk(kid, seen)
        }
}

                pairs = map[string]Literal{}
        }
        return &Obj{Pairs: pairs}
}

func (*Obj) Type() ValueType {
        return OBJ_VAL

func (o *Obj) String() string {
        if HasCycle(o) {
                return circularReference
        }
        return o.stringNoCycleCheck()
}

// stringNoCycleCheck is the String() without the HasCycle check.
// It builds the output from the elements, and for each element it skips
// the cycle check for child arrays and objects.
func (o *Obj) stringNoCycleCheck() string {
        return o.joinPairs(writeStringPair)
}

// htmlNoCycleCheck is stringNoCycleCheck with HTML escaping, see escapedString.
func (o *Obj) htmlNoCycleCheck() string {
        return o.joinPairs(writeHTMLPair)
}

// joinPairs writes each key and value with writePair, sorted by key.
func (o *Obj) joinPairs(writePair func(out *strings.Builder, key string, val Literal)) string {
        if o.Pairs == nil {
                return "{}"
        }

        keys := o.sortedKeys()
        var out strings.Builder
        out.Grow(2)

        out.WriteByte('{')

        for i, key := range keys {
                if i > 0 {
                        out.WriteString(", ")
                }
                writePair(&out, key, o.Pairs[key])
        }

        out.WriteByte('}')

        return out.String()
}

// writeStringPair writes `key: "string"` for strings and `key: value` for
// other values.
func writeStringPair(out *strings.Builder, key string, val Literal) {
        out.WriteString(key)
        out.WriteString(": ")

        if str, isStr := val.(*Str); isStr {
                out.WriteString(`"`)
                out.WriteString(str.Val)
                out.WriteString(`"`)
                return
        }

        out.WriteString(stringOf(val))
}

// writeHTMLPair is writeStringPair with HTML escaping. The key is always
// escaped. A raw string is written with its quotes as is, other strings get
// their quotes escaped as &#34;, like html.EscapeString does.
func writeHTMLPair(out *strings.Builder, key string, val Literal) {
        out.WriteString(html.EscapeString(key))
        out.WriteString(": ")

        str, isStr := val.(*Str)

        switch {
        case isStr && str.raw:
                out.WriteString(`"`)
                out.WriteString(str.Val)
                out.WriteString(`"`)
        case isStr:
                out.WriteString("&#34;")
                out.WriteString(html.EscapeString(str.Val))
                out.WriteString("&#34;")
        default:
                out.WriteString(htmlOf(val))
        }
}

func (o *Obj) JSON() (string, error) {
        if HasCycle(o) {
                return "", fmt.Errorf(fail.ErrCircularValue, o.Type())
        }
        return o.jsonNoCycleCheck()
}

func (o *Obj) jsonNoCycleCheck() (string, error) {
        if o.Pairs == nil {
                return "{}", nil
        }

        keys := o.sortedKeys()
        var out strings.Builder
        out.Grow(2)

        out.WriteByte('{')

        for i, k := range keys {
                pair := o.Pairs[k]
                if i > 0 {
                        out.WriteByte(',')
                }

                encodedKey, err := marshalJSONString(k)
                if err != nil {
                        return "", err
                }

                jsonVal, err := jsonOf(pair)
                if err != nil {
                        return "", err
                }

                out.WriteString(encodedKey)
                out.WriteByte(':')
                out.WriteString(jsonVal)
        }

        out.WriteByte('}')

        return out.String(), nil
}

func (o *Obj) Dump(ident int) string {
        if HasCycle(o) {
                return circularReference
        }
        return o.dumpNoCycleCheck(ident)
}

func (o *Obj) dumpNoCycleCheck(ident int) string {
        if o.Pairs == nil {
                return "{}"
        }

        spaces := strings.Repeat("  ", ident)
        ident += 1

        var out strings.Builder
        out.Grow(4)

        fmt.Fprintf(&out, `<span style="%s">object:%d </span>`, DUMP_META, len(o.Pairs))
        fmt.Fprintf(&out, `<span style="%s">{</span>`, DUMP_BRACE)

        if len(o.Pairs) == 0 {
                spaces = ""
        } else {
                out.WriteByte('\n')
        }

        insideSpaces := strings.Repeat("  ", ident)

        for key, pair := range o.Pairs {
                out.WriteString(insideSpaces)
                fmt.Fprintf(&out, `<span style="%s">"`, DUMP_PROP)
                out.WriteString(html.EscapeString(key))
                fmt.Fprintf(&out, `"</span>: `)
                out.WriteString(dumpOf(pair, ident))
                out.WriteString(",\n")
        }

        out.WriteString(spaces)
        fmt.Fprintf(&out, `<span style="%s">}</span>`, DUMP_BRACE)

        return out.String()
}

        if HasCycle(o) {
                return nil
        }
        return o.nativeNoCycleCheck()
}

        res := map[string]any{}
        for k, v := range o.Pairs {
                res[k] = nativeOf(v)
        }
        return res
}

// Raw returns v with every string in it marked as raw, so it's printed
// without HTML escaping. Arrays and objects are traversed recursively and
// copied, so v itself is never changed. Other literal types are returned as is.
func Raw(v Literal) Literal {
        return rawCopy(v, map[Literal]Literal{})
}

// rawCopy returns v with its strings marked as raw. The copies map stores
// each array and object that is already copied, with the original as the key
// and its copy as the value.
func rawCopy(v Literal, copies map[Literal]Literal) Literal {
        switch v := v.(type) {
        case *Str:
                return rawCopyStr(v)
        case *Arr:
                return rawCopyArr(v, copies)
        case *Obj:
                return rawCopyObj(v, copies)
        default:
                return v
        }
}

// rawCopyStr returns s if it is already raw, otherwise a raw copy of it.
func rawCopyStr(s *Str) *Str {
        if s.raw {
                return s
        }
        return &Str{Val: s.Val, raw: true}
}

// rawCopyArr returns a deep copy of arr with its strings marked as raw.
// If arr is already in copies, its stored copy is returned. The new copy is
// stored before its elements are copied, so an array that contains itself
// resolves to its copy instead of looping forever.
func rawCopyArr(arr *Arr, copies map[Literal]Literal) *Arr {
        if stored, ok := copies[arr]; ok {
                return stored.(*Arr)
        }

        arrayCopy := &Arr{Elements: make([]Literal, len(arr.Elements))}
        copies[arr] = arrayCopy

        for i, elem := range arr.Elements {
                arrayCopy.Elements[i] = rawCopy(elem, copies)
        }

        return arrayCopy
}

// rawCopyObj is rawCopyArray for objects.
func rawCopyObj(obj *Obj, copies map[Literal]Literal) *Obj {
        if stored, ok := copies[obj]; ok {
                return stored.(*Obj)
        }

        objectCopy := NewObj(make(map[string]Literal, len(obj.Pairs)))
        copies[obj] = objectCopy

        for key, val := range obj.Pairs {
                objectCopy.Pairs[key] = rawCopy(val, copies)
        }

        return objectCopy
}

type Scope struct {
        vars   map[string]Literal
        parent *Scope
}

// NewScope returns a scope with global value.
// A nil global falls back to the shared empty one.
func NewScope(global *Obj) *Scope {
        if global == nil {
                global = emptyGlobal
        }

        return &Scope{vars: map[string]Literal{"global": global}}
}

func NewChildScope(parent *Scope) *Scope {
        return &Scope{vars: map[string]Literal{}, parent: parent}
}

func NewScopeFromMap(data map[string]any, global *Obj) (*Scope, *fail.Error) {
        scope := NewScope(global)

        for key, val := range data {
                obj := NativeToValue(val)
                if errObj, ok := obj.(*Error); ok {
                        return nil, errObj.Err
                }

                if err := scope.Set(key, obj); err != nil {
                        return nil, fail.New(nil, "", fail.OriginEval, "%s", err.Error())
                }
        }

        return scope, nil
}

func (e *Scope) Get(name string) (Literal, bool) {
        if val, ok := e.vars[name]; ok {
                return val, true
        }

        if e.parent != nil {
                return e.parent.Get(name)
        }

        return nil, false
}

func (e *Scope) Set(key string, val Literal) error {
        if key == "loop" || key == "global" {
                return errors.New(fail.ErrReservedIdentifiers)
        }

        if oldVar, ok := e.isTypeMismatch(key, val); ok {
                return e.identifierMismatchError(key, oldVar, val)
        }

        writeScope := e.scopeOf(key)
        if writeScope == nil {
                writeScope = e
        }

        writeScope.vars[key] = val

        return nil
}

// scopeOf walks up from the current scope and returns the closest one
// that defines key. It returns nil when no scope in the chain defines it.
func (e *Scope) scopeOf(key string) *Scope {
        if _, ok := e.vars[key]; ok {
                return e
        }

        if e.parent == nil {
                return nil
        }

        return e.parent.scopeOf(key)
}

func (e *Scope) SetLoopVariable(key string, val Literal) error {
        if key == "loop" || key == "global" {
                return errors.New(fail.ErrReservedIdentifiers)
        }

        e.vars[key] = val
        return nil
}

func (e *Scope) SetLoopVar(pairs map[string]Literal) {
        e.vars["loop"] = NewObj(pairs)
}

func (e *Scope) isTypeMismatch(key string, val Value) (Value, bool) {
        oldVar, ok := e.Get(key)
        return oldVar, (ok && oldVar != nil && oldVar.Type() != val.Type())
}

func (e *Scope) identifierMismatchError(key string, oldVar, val Value) error {
        msg := fmt.Sprintf(fail.ErrIdentTypeMismatch, key, oldVar.Type(), val.Type())
        return errors.New(msg)
}

// refKey identifies a slice, map or pointer that is currently being
// converted. It is used to detect cycles in native maps, slices and pointers.
type refKey struct {

        ptr uintptr
        len int
}

// IsControlDir reports whether the value is a break or continue directive.
func IsControlDir(val Value) bool {
        return val.Is(BREAK_VAL) || val.Is(CONTINUE_VAL)
}

// HasBreak reports whether the value is a break, or a block holding one.
func HasBreak(val Value) bool {
        if block, ok := val.(*Block); ok {
                return block.HasBreak()
        }

        return val.Is(BREAK_VAL)
}

// HasContinue reports whether the value is a continue, or a block holding one.
func HasContinue(val Value) bool {
        if block, ok := val.(*Block); ok {
                return block.HasContinue()
        }

        return val.Is(CONTINUE_VAL)

func NativeToValue(val any) Literal {
        if val == nil {
                return new(Nil)
        }

        lit, failure := reflectToLiteral(reflect.ValueOf(val), "", map[refKey]bool{})
        if failure != nil {
                return &Error{Err: failure}
        }

        return lit
}

func circularReferenceError(v reflect.Value) *fail.Error {
        return fail.New(nil, "", fail.OriginTpl, fail.ErrCircularValue, v.Type())
}

// reflectToLiteral converts v into a Literal. The field is the struct field
// name the value came from, used for error messages. It's empty for values
// that aren't struct fields.
func reflectToLiteral(
        v reflect.Value,
        field string,
        seen map[refKey]bool,
) (Literal, *fail.Error) {
        if !v.IsValid() {
                return new(Nil), nil
        }

        if key, ok := refKeyFor(v); ok {
                if seen[key] {
                        return nil, circularReferenceError(v)
                }

                seen[key] = true
                defer delete(seen, key)
        }

        switch v.Kind() {
        case reflect.String:
                return &Str{Val: v.String()}, nil
        case reflect.Bool:
                return &Bool{Val: v.Bool()}, nil
        case reflect.Int, reflect.Int8, reflect.Int16, reflect.Int32, reflect.Int64:
                return &Int{Val: v.Int()}, nil
        case reflect.Uint, reflect.Uint8, reflect.Uint16, reflect.Uint32, reflect.Uint64:
                return uintToInt(v, field)
        case reflect.Float32, reflect.Float64:
                return &Float{Val: v.Float()}, nil
        case reflect.Struct:
                if t, ok := v.Interface().(time.Time); ok {
                        return &Str{Val: t.Format("2006-01-02 15:04:05")}, nil
                }

                return nativeStructToValue(v, seen)
        case reflect.Slice, reflect.Array:
                return nativeSliceToValue(v, seen)
        case reflect.Map:
                return nativeMapToValue(v, seen)
        case reflect.Pointer:
                if v.IsNil() {
                        return new(Nil), nil
                }

                return reflectToLiteral(v.Elem(), field, seen)
        case reflect.Interface:
                if v.IsNil() {
                        return new(Nil), nil
                }

                return reflectToLiteral(v.Elem(), field, seen)
        }

        return nil, unsupportedTypeError(v, field)
}

// refKeyFor returns the refKey for slices, maps and non-nil pointers. Nil
// pointers are skipped because they have no element to recurse into.
func refKeyFor(v reflect.Value) (refKey, bool) {
        switch v.Kind() {
        case reflect.Slice:
                return refKey{typ: v.Type(), ptr: v.Pointer(), len: v.Len()}, true
        case reflect.Map:
                return refKey{typ: v.Type(), ptr: v.Pointer()}, true
        case reflect.Pointer:
                return refKey{typ: v.Type(), ptr: v.Pointer()}, !v.IsNil()
        }

        return refKey{}, false
}

func uintToInt(v reflect.Value, field string) (Literal, *fail.Error) {
        val := v.Uint()
        if val > math.MaxInt64 {
                return nil, uintOverflowError(val, field)
        }

        return &Int{Val: int64(val)}, nil
}

func unsupportedTypeError(v reflect.Value, field string) *fail.Error {
        if field == "" {
                return fail.New(nil, "", fail.OriginTpl, fail.ErrUnsupportedType, v.Type())
        }

        return fail.New(nil, "", fail.OriginTpl, fail.ErrUnsupportedFieldType, v.Type(), field)
}

func uintOverflowError(val uint64, field string) *fail.Error {
        if field == "" {
                return fail.New(nil, "", fail.OriginTpl, fail.ErrUintOverflow, val)
        }

        return fail.New(nil, "", fail.OriginTpl, fail.ErrUintOverflowField, val, field)
}

func nativeMapToValue(v reflect.Value, seen map[refKey]bool) (Literal, *fail.Error) {
        obj := NewObj(nil)

        iter := v.MapRange()
        for iter.Next() {
                key := mapKeyToString(iter.Key())

                value, err := reflectToLiteral(iter.Value(), "", seen)
                if err != nil {
                        return nil, err
                }

                obj.Pairs[key] = value
        }

        return obj, nil
}

// mapKeyToString converts any map key into the string form used by Obj pairs.
func mapKeyToString(k reflect.Value) string {
        return fmt.Sprint(k.Interface())
}

func nativeStructToValue(v reflect.Value, seen map[refKey]bool) (Literal, *fail.Error) {
        obj := NewObj(nil)
        valType := v.Type()

        for i := 0; i < v.NumField(); i++ {
                field := valType.Field(i)
                if !field.IsExported() {
                        continue
                }

                lit, failure := reflectToLiteral(v.Field(i), field.Name, seen)
                if failure != nil {
                        return nil, failure
                }

                obj.Pairs[field.Name] = lit
        }

        return obj, nil
}

func nativeSliceToValue(v reflect.Value, seen map[refKey]bool) (Literal, *fail.Error) {
        arr := new(Arr)
        arr.Elements = make([]Literal, v.Len())

        for i := 0; i < v.Len(); i++ {
                lit, failure := reflectToLiteral(v.Index(i), "", seen)
                if failure != nil {
                        return nil, failure
                }

                arr.Elements[i] = lit
        }

        return arr, nil
}

// marshalJSONString encodes s as a JSON string without HTML escaping. Textwire
// is a text engine and {!! !!} is the explicit no-escaping operator, so .json()
// doesn't escape HTML characters like <, > and &.
func marshalJSONString(s string) (string, error) {
        var buf bytes.Buffer

        enc := json.NewEncoder(&buf)
        enc.SetEscapeHTML(false)

        if err := enc.Encode(s); err != nil {
                return "", err
        }

        // Encode writes a trailing newline after the value, we remove it
        return strings.TrimSuffix(buf.String(), "\n"), nil
}

const (
        tplNotFoundPrefix = "template file '"
        tplNotFoundSuffix = "' not found"
)

// Template holds all necessary data which it will use when individual
// template files will be evaluated by String() or Response() methods.
type Template struct {
        linker *linker.NodeLinker
        global *value.Obj
}

// NewTemplate returns a new Template instance with parsed Textwire files
// provided by configuration options. The Template instance should be used
// for evaluating Textwire in your handlers.
func NewTemplate(opt *config.Config) (*Template, *fail.Error) {
        Configure(opt)

        global, failure := value.NewFrozenGlobal(userConf.GlobalData)
        if failure != nil {
                return nil, failure
        }

        files, err := locateFiles()
        if err != nil {
                return nil, fail.FromError(err, nil, "", fail.OriginTpl)
        }

        programs, parseErr := parseFiles(files)
        if parseErr != nil {
                return nil, parseErr
        }

        ln := linker.New(programs)
        if failure := ln.LinkNodes(); failure != nil {
                return nil, failure
        }

        tpl := &Template{
                linker: ln,
                global: global,
        }

        if userConf.FileWatcher {
                newFileWatcher(ln).Watch()
        }

        return tpl, nil
}

// String returns final evaluated template result represented as a string.
func (t *Template) String(name string, data map[string]any) (string, *fail.Error) {
        t.linker.RLock()
        linkErr, progs := t.linker.LinkError, t.linker.Programs
        t.linker.RUnlock()

        if linkErr != nil {
                return "", linkErr
        }

        scope, err := value.NewScopeFromMap(data, t.global)
        if err != nil {
                return "", err
        }

        name = file.ReplacePathAlias(name, file.PathAliasViews)
        prog := ast.FindProg(name, progs)
        if prog == nil {
                relPath := file.NameToRelPath(name, userConf.TemplateDir, userConf.TemplateExt)
                return "", fail.New(nil, relPath, fail.OriginTpl, fail.ErrTemplateNotFound, name)
        }

        e := evaluator.New(customFunc, userConf, t.global)
        ctx := evaluator.NewContext(scope, prog.AbsPath)
        evaluated := e.Eval(prog, ctx)
        if evaluated.Is(value.ERR_VAL) {
                return "", evaluated.(*value.Error).Err
        }

        return evaluated.String(), nil
}

// Response evaluates template file with String() method and passing that final
// string to the given http.ResponseWriter.
func (t *Template) Response(w http.ResponseWriter, name string, data map[string]any) *fail.Error {
        evaluated, failure := t.String(name, data)
        if failure == nil {
                return writeBody(w, name, evaluated)
        }

        // Rendering the error response itself failed: report that error instead.
        if err := t.writeFailure(w, name, failure); err != nil {
                return err
        }

        return failure
}

// isTemplateNotFound reports whether the failure is caused by a missing
// template file.
func isTemplateNotFound(failure *fail.Error) bool {
        if failure == nil || failure.Origin() != fail.OriginTpl {
                return false
        }

        msg := failure.Message()
        return strings.HasPrefix(msg, tplNotFoundPrefix) &&
                strings.HasSuffix(msg, tplNotFoundSuffix)
}

// httpStatusFor maps a template failure to the HTTP status code to respond with.
func httpStatusFor(failure *fail.Error) int {
        if isTemplateNotFound(failure) {
                return http.StatusNotFound
        }

        return http.StatusInternalServerError
}

// writeBody writes the given content to w, wrapping any write error.
func writeBody(w http.ResponseWriter, name, content string) *fail.Error {
        if _, err := fmt.Fprint(w, content); err != nil {
                return fail.FromError(err, nil, name, fail.OriginTpl)
        }

        return nil
}

// writeFailure writes the error response (custom error page or the built-in
// one) for the given failure, using the appropriate HTTP status.
func (t *Template) writeFailure(
        w http.ResponseWriter,
        name string,
        failure *fail.Error,
) *fail.Error {
        status := httpStatusFor(failure)

        if userConf.ErrorPagePath != "" && !userConf.DebugMode {
                w.WriteHeader(status)
                return t.responseCustomErrorPage(w)
        }

        page, err := errorPage(failure)
        if err != nil {
                return err
        }

        w.WriteHeader(status)
        return writeBody(w, name, page)
}

func (t *Template) responseCustomErrorPage(w http.ResponseWriter) *fail.Error {
        evaluated, failure := t.String(userConf.ErrorPagePath, nil)
        if failure != nil {
                return failure
        }

        _, err := fmt.Fprint(w, evaluated)
        if err != nil {
                return fail.FromError(err, nil, userConf.ErrorPagePath, fail.OriginTpl)
        }

        return nil
}

// EvaluateString evaluates a given inp string containing Textwire code.
// The function accepts a string template and data to inject into Textwire.
// After evaluation, it returns the processed string and any error encountered.
func EvaluateString(inp string, data map[string]any) (string, *fail.Error) {
        prog, errs := parseStr(inp)
        if len(errs) != 0 {
                return "", errs[0]
        }

        scope, err := value.NewScopeFromMap(data, nil)
        if err != nil {
                return "", err
        }

        e := evaluator.New(customFunc, nil, nil)
        ctx := evaluator.NewContext(scope, prog.AbsPath)
        evaluated := e.Eval(prog, ctx)
        if evaluated.Is(value.ERR_VAL) {
                return "", evaluated.(*value.Error).Err
        }

        return evaluated.String(), nil
}

// EvaluateFile evaluates a file containing Textwire code.
//
// The absPath an absolute path to the Textwire file.
// The data is a map of variables you want to inject into the Textwire.

        f := file.New("", "", absPath, userConf)
        content, err := f.Content()
        if err != nil {
                return "", fail.FromError(err, nil, absPath, fail.OriginTpl)
        }

        res, failure := EvaluateString(content, data)
        if failure != nil {
                return "", failure
        }

        return res, nil
}

//go:embed embed/default-error-page.tw
var defaultErrPage string

// errorPage returns HTML that's displayed when an error occurs while
// rendering template.
func errorPage(failure *fail.Error) (string, *fail.Error) {
        data := map[string]any{
                "path":      failure.Filepath(),
                "line":      failure.Pos().Line(),
                "col":       failure.Pos().Col(),
                "message":   failure.Message(),
                "debugMode": userConf.DebugMode,
        }

        out, err := EvaluateString(defaultErrPage, data)
        if err != nil {
                return "", err
        }

        return out, nil
}

func parseStr(text string) (*ast.Program, []*fail.Error) {
        l := lexer.New(text)
        p := parser.New(l, nil)

        prog := p.ParseProgram()
        if p.HasErrors() {
                return nil, p.Errors()
        }

        return prog, nil
}

// parseFiles parses each Textwire file into AST nodes and returns them.
func parseFiles(files []*file.SourceFile) ([]*ast.Program, *fail.Error) {
        programs := make([]*ast.Program, 0, len(files))
        for _, f := range files {
                prog, failure, parseErr := parseFile(f)
                if parseErr != nil {
                        return programs, fail.FromError(parseErr, nil, f.Abs, fail.OriginTpl)
                }

                if failure != nil {
                        return programs, failure
                }

                programs = append(programs, prog)
        }

        return programs, nil
}

// parseFile parses given file into a ast.Program and returns it.
func parseFile(f *file.SourceFile) (*ast.Program, *fail.Error, error) {
        content, err := f.Content()
        if err != nil {
                return nil, nil, err
        }

        prog, failure := parseContent(f, content)
        return prog, failure, nil
}

// parseContent parses given content of the file into a ast.Program and returns it.
func parseContent(f *file.SourceFile, content string) (*ast.Program, *fail.Error) {
        l := lexer.New(content)
        p := parser.New(l, f)
        if p.HasErrors() {
                return nil, p.Errors()[0]
        }

        prog := p.ParseProgram()
        prog.AbsPath = f.Abs
        prog.Name = f.Name

        if p.HasErrors() {
                return nil, p.Errors()[0]
        }

        return prog, nil
}

// makeFileNameSet returns a set of file names for quick lookup.
func makeFileNameSet(files []*file.SourceFile) map[string]bool {
        set := make(map[string]bool, len(files))
        for _, f := range files {
                set[f.Name] = true
        }
        return set
}

// locateFiles recursively finds all Textwire files in the templates directory,
// creates a *file.SourceFile wrapper for each of them, and returns the
// discovered files.
func locateFiles() ([]*file.SourceFile, error) {
        var files []*file.SourceFile

        walk := func(path string, d fs.DirEntry, err error) error {
                if err != nil {
                        return err
                }

                if !isTemplateFile(path, d) {
                        return nil
                }

                sf, err := newSourceFile(path, d)
                if err != nil {
                        return err
                }

                files = append(files, sf)
                return nil
        }

        if err := fs.WalkDir(userConf.TemplateFS, ".", walk); err != nil {
                return nil, err
        }

        return files, nil
}

// isTemplateFile reports whether the entry is a regular file with the
// configured template extension.
func isTemplateFile(path string, d fs.DirEntry) bool {
        return !d.IsDir() && strings.HasSuffix(path, userConf.TemplateExt)
}

// newSourceFile builds a SourceFile for the template at the given walk path.
func newSourceFile(path string, d fs.DirEntry) (*file.SourceFile, error) {
        // When templates are embedded via config.TemplateFS, the walked path
        // already contains config.TemplateDir, so strip it to avoid duplication.
        if userConf.UsesFS() {
                path = strings.Replace(path, userConf.TemplateDir, "", 1)
        }

        relPath := file.JoinPaths(userConf.TemplateDir, path)
        absPath, err := filepath.Abs(relPath)
        if err != nil {
                return nil, err
        }

        info, err := d.Info()
        if err != nil {
                return nil, err
        }

        name := strings.TrimSuffix(path, userConf.TemplateExt)
        sf := file.New(name, relPath, absPath, userConf)
        sf.ModTime = info.ModTime()

        return sf, nil
}

import (

        "path/filepath"
        "slices"
        "strings"
        "time"

        "github.com/textwire/textwire/v6/pkg/ast"
        "github.com/textwire/textwire/v6/pkg/fail"
        "github.com/textwire/textwire/v6/pkg/file"
        "github.com/textwire/textwire/v6/pkg/linker"
)

// fileWatcher monitors template files for changes and refreshes parsed AST nodes.
// It is designed for development use only due to performance implications.
type fileWatcher struct {
        linker    *linker.NodeLinker
        logger    *WatcherLogger
        ticker    *time.Ticker
        files     []*file.SourceFile
        sources   map[string]string // last valid source of each template, by name
        lastError string
}

// newFileWatcher creates a new file watcher instance.
func newFileWatcher(oldLinker *linker.NodeLinker) *fileWatcher {
        return &fileWatcher{
                linker:  oldLinker,
                logger:  NewWatcherLogger(),
                sources: make(map[string]string),
        }
}

// Watch starts monitoring files in a background goroutine.
// It detects file creation, deletion, and modifications, then reparses and
// relinks accordingly.
func (fw *fileWatcher) Watch() {
        if userConf.UsesFS() {
                fw.logger.Fatal("cannot use config.FileWatcher when using config.TemplateFS")
        }

        fw.logger.Info("watching files for changes...")

        var err error
        fw.files, err = locateFiles()
        if err != nil {
                fw.logger.Fatal("error locating files " + err.Error())
        }

        fw.ticker = time.NewTicker(userConf.WatcherInterval)

        go func() {
                for range fw.ticker.C {
                        fw.tick()
                }
        }()
}

// tick runs on every tick. It rescans the template dir when files
// are added or deleted, then relinks everything so changed templates and link
// errors are tracked.
func (fw *fileWatcher) tick() {
        if fw.filesAddedOrDeleted() {
                fw.rescanTemplateDir()
        }

        fw.relinkAllPrograms()
}

// filesAddedOrDeleted reports whether files were added or deleted on disk.
func (fw *fileWatcher) filesAddedOrDeleted() bool {
        return fw.countFiles() != len(fw.files) || slices.ContainsFunc(fw.files, fw.fileWasDeleted)
}

// rescanTemplateDir re-locates template files and removes programs of files
// that were deleted.
func (fw *fileWatcher) rescanTemplateDir() {
        fw.logger.Info("file count changed, updating...")
        oldFiles := fw.files

        files, err := locateFiles()
        if err != nil {
                fw.logger.Error("error locating files " + err.Error())
                return
        }

        fw.files = files
        fw.removeDeletedPrograms(oldFiles)
}

func (fw *fileWatcher) fileWasDeleted(f *file.SourceFile) bool {
        _, err := os.Stat(f.Abs)
        return os.IsNotExist(err)
}

// relinkAllPrograms parses the current files into fresh programs, links them, and
// swaps the whole slice under the write lock. Existing programs are never
// mutated, so renders that hold older programs stay consistent.
func (fw *fileWatcher) relinkAllPrograms() {
        if fw.files == nil {
                return
        }

        programs := make([]*ast.Program, 0, len(fw.files))
        for _, f := range fw.files {
                if prog := fw.parseFileOrLastValid(f); prog != nil {
                        programs = append(programs, prog)
                }
        }

        ln := linker.New(programs)
        failure := ln.LinkNodes()

        fw.withLock(func() {
                fw.linker.Programs = ln.Programs
                fw.trackLinkingError(failure)
        })
}

// parseFileOrLastValid parses f into a fresh program. When f fails to parse, it
// parses the last valid source of f instead, so one broken template doesn't
// block the rest. Returns nil when f has never parsed.
func (fw *fileWatcher) parseFileOrLastValid(f *file.SourceFile) *ast.Program {
        content, err := f.Content()
        if err != nil {
                fw.logger.Error(err.Error())
                return fw.parseLastValid(f)
        }

        changed := fw.trackModTime(f)

        prog, failure := parseContent(f, content)
        if failure != nil {
                fw.logger.Error(failure.Error().Error())
                return fw.parseLastValid(f)
        }

        fw.sources[f.Name] = content

        if changed {
                fw.logger.Info("refreshed " + f.Rel)
        }

        return prog
}

// trackModTime updates passed file's modification time and reports whether
// the file changed since the previous tick.
func (fw *fileWatcher) trackModTime(f *file.SourceFile) bool {
        info, err := os.Stat(f.Abs)
        if err != nil {
                return false
        }

        changed := !f.ModTime.IsZero() && !info.ModTime().Equal(f.ModTime)
        f.ModTime = info.ModTime()

        return changed
}

// parseLastValid parses the last valid source of f into a fresh program.
// Returns nil when f has never parsed.
func (fw *fileWatcher) parseLastValid(f *file.SourceFile) *ast.Program {
        content, ok := fw.sources[f.Name]
        if !ok {
                return nil
        }

        prog, _ := parseContent(f, content)
        return prog
}

// trackLinkingError logs linking errors once and stores them for Template access.
func (fw *fileWatcher) trackLinkingError(failure *fail.Error) {
        fw.linker.LinkError = failure

        if failure == nil {
                if fw.lastError != "" {
                        fw.logger.Info("all templates are valid!")
                        fw.lastError = ""
                }
                return
        }

        errMsg := failure.Error().Error()
        if errMsg == fw.lastError {
                return
        }

        fw.logger.Error(errMsg)
        fw.lastError = errMsg
}

// removeDeletedPrograms removes programs and cached sources of files that
// existed in oldFiles but are no longer in fw.files.
func (fw *fileWatcher) removeDeletedPrograms(oldFiles []*file.SourceFile) {
        for _, f := range fw.findDeletedFiles(oldFiles) {
                fw.logger.Info("removed " + f.Rel)
                delete(fw.sources, f.Name)
                fw.removeProgramByName(f.Name)
        }
}

// findDeletedFiles returns files from oldFiles that are not in fw.files.
func (fw *fileWatcher) findDeletedFiles(oldFiles []*file.SourceFile) []*file.SourceFile {
        current := makeFileNameSet(fw.files)

        var deleted []*file.SourceFile
        for _, f := range oldFiles {
                if !current[f.Name] {
                        deleted = append(deleted, f)
                }
        }

        return deleted
}

// removeProgramByName removes the program with the specified name from the linker.
func (fw *fileWatcher) removeProgramByName(name string) {
        fw.filterPrograms(func(p *ast.Program) bool {
                return p.Name != name
        })
}

// filterPrograms replaces linker programs with those for which keep returns true.
func (fw *fileWatcher) filterPrograms(keep func(*ast.Program) bool) {
        fw.withLock(func() {
                programs := make([]*ast.Program, 0, len(fw.linker.Programs))
                for _, p := range fw.linker.Programs {
                        if keep(p) {
                                programs = append(programs, p)
                        }
                }

                fw.linker.Programs = programs
        })
}

// withLock executes the given function while holding the linker's write lock.
func (fw *fileWatcher) withLock(fn func()) {
        fw.linker.Lock()
        defer fw.linker.Unlock()
        fn()
}

// countFiles counts template files in the templates directory on disk.
func (fw *fileWatcher) countFiles() int {
        count := 0
        err := filepath.WalkDir(
                userConf.TemplateDir,
                func(path string, d os.DirEntry, err error) error {
                        if err == nil && !d.IsDir() && strings.HasSuffix(path, userConf.TemplateExt) {
                                count++
                        }
                        return nil
                },
        )

        if err != nil {
                fw.logger.Error("error counting files " + err.Error())
        }

        return count
}
